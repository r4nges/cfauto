# CFAUTO — Sub-projeto 1: Fundação — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Entregar o esqueleto executável do CFAUTO: banco SQLite com migrations, usuários com login e permissões por módulo, licença offline Ed25519, provisionamento na primeira execução, auditoria, jobs agendados, janela principal com navegação e empacotamento PyInstaller.

**Architecture:** Monolito em camadas (spec §2). Pacote `cfauto/` com `core/` (infra sem regra de domínio), domínios (`auth/`, `licenca/`, `sistema/`, `jobs/`) com `models.py`/`service.py`, e `ui/` (PySide6) que só chama serviços. Serviços recebem `Session` + `Ator` e verificam permissão. Escrita sempre em `transacao()`, com auditoria na mesma transação.

**Tech Stack:** Python 3.13 (uv, Python gerenciado), PySide6, SQLAlchemy 2.x, Alembic, APScheduler 3.x, argon2-cffi, cryptography (Ed25519), python-dateutil; pytest, pytest-qt, pytest-cov, ruff, mypy, pre-commit, PyInstaller.

**Spec:** `docs/superpowers/specs/2026-09-23-cfauto-arquitetura-design.md` (seções 2, 3.7, 6, 7). Leia a spec antes de começar.

## Global Constraints

- Idioma do código de domínio, mensagens e UI: português (identificadores sem acento: `usuario`, `transacao`). Mensagens ao usuário com acentuação correta.
- Python 3.13 via uv com `python-preference = "only-managed"` (a spec cita 3.12; 3.13 é o disponível e suportado pelo PySide6 — divergência registrada).
- Dinheiro e quantidade: `Decimal`, nunca `float`. Colunas monetárias usam `Dinheiro()` (inteiro em centavos) e quantidade `Quantidade()` (inteiro em milésimos) — em SQLite, `Numeric` perde precisão; o tipo escalado mantém a semântica `Numeric(12,2)`/`Numeric(12,3)` da spec com soma exata em SQL.
- Data atual só por `cfauto.core.clock.agora()` / `clock.hoje()`. Nunca `datetime.now()` fora de `clock.py`.
- Datas externas passam por `cfauto.core.dates` na borda; string de data não circula internamente.
- Acesso a dados sempre por nome de atributo; proibido indexar tuplas/listas posicionais para campos de negócio.
- `ui/` nunca importa `sqlalchemy` nem modelos para consultar; chama funções de `service`.
- Nenhuma exceção é engolida devolvendo valor padrão; erros de regra são subclasses de `cfauto.core.errors.ErroDominio`.
- Banco em `%APPDATA%\CFAUTO\cfauto.db` (sobrescrevível por `CFAUTO_HOME`); backups em `%APPDATA%\CFAUTO\backups` (7 últimos); logs em `%APPDATA%\CFAUTO\logs`.
- Senhas: argon2id; 5 falhas → bloqueio 5 min; senha mínima 6 caracteres; bloqueio por inatividade padrão 15 min.
- Licença: `base64url(payload_json) + "." + base64url(assinatura_ed25519)`; payload `{cliente, emitida, expira, maquina?}`; só a chave pública no app; anti-relógio bloqueia se `hoje < ultima_data_vista − 1 dia`.
- Commits: Conventional Commits, terminando com as linhas de attribution vigentes da sessão.
- Comandos rodam na raiz `C:\Users\Admin\Documents\CFAUTO` com `uv run ...`.

```python
from collections.abc import Callable
from datetime import date, datetime

_fonte: Callable[[], datetime] = datetime.now


def agora() -> datetime:
    return _fonte()


def hoje() -> date:
    return _fonte().date()


def definir(fonte: Callable[[], datetime]) -> None:
    global _fonte
    _fonte = fonte


def resetar() -> None:
    definir(datetime.now)
```

```python
"""Dinheiro sempre em Decimal. Parser aceita formato brasileiro."""

import re
from decimal import ROUND_HALF_UP, Decimal, InvalidOperation

from cfauto.core.errors import ValorInvalido

CENTAVO = Decimal("0.01")
_MILHAR_BR = re.compile(r"^\d{1,3}(\.\d{3})+$")


def arredondar(valor: Decimal) -> Decimal:
    return valor.quantize(CENTAVO, rounding=ROUND_HALF_UP)


def para_decimal(valor: str | int | float | Decimal) -> Decimal:
    if isinstance(valor, bool):
        raise ValorInvalido(f"Valor inválido: {valor!r}")
    if isinstance(valor, Decimal):
        return valor
    if isinstance(valor, int):
        return Decimal(valor)
    if isinstance(valor, float):
        return Decimal(str(valor))
    texto = valor.strip().replace("\u00a0", "").replace(" ", "")
    negativo = texto.startswith("-")
    if negativo:
        texto = texto[1:]
    texto = texto.replace("R$", "")
    if not texto:
        raise ValorInvalido("Valor vazio.")
    if "," in texto:
        texto = texto.replace(".", "").replace(",", ".")
    elif _MILHAR_BR.match(texto):
        texto = texto.replace(".", "")
    try:
        resultado = Decimal(texto)
    except InvalidOperation as erro:
        raise ValorInvalido(f"Valor inválido: {valor!r}") from erro
    if not resultado.is_finite():
        raise ValorInvalido(f"Valor inválido: {valor!r}")
    return -resultado if negativo else resultado


def formatar_brl(valor: Decimal) -> str:
    v = arredondar(valor)
    sinal = "-" if v < 0 else ""
    inteiro, _, centavos = f"{abs(v):.2f}".partition(".")
    grupos = f"{int(inteiro):,}".replace(",", ".")
    return f"{sinal}R$ {grupos},{centavos}"
```

```python
"""Normalização de datas na borda (UI, importador, XML, APIs)."""

from datetime import date, datetime

from cfauto.core.errors import DataInvalida

_FORMATOS_DATA = ("%d/%m/%Y", "%Y-%m-%d", "%d/%m/%y", "%d-%m-%Y")
_FORMATOS_DATAHORA = (
    "%d/%m/%Y %H:%M:%S",
    "%d/%m/%Y %H:%M",
    "%Y-%m-%d %H:%M:%S",
    "%Y-%m-%d %H:%M",
    "%Y-%m-%dT%H:%M:%S",
    "%Y-%m-%dT%H:%M:%S.%f",
)


def para_datahora(valor: date | datetime | str) -> datetime:
    if isinstance(valor, datetime):
        return valor
    if isinstance(valor, date):
        return datetime(valor.year, valor.month, valor.day)
    if isinstance(valor, str):
        texto = valor.strip()
        for formato in _FORMATOS_DATAHORA + _FORMATOS_DATA:
            try:
                return datetime.strptime(texto, formato)
            except ValueError:
                continue
    raise DataInvalida(f"Data inválida: {valor!r}")


def para_data(valor: date | datetime | str) -> date:
    if isinstance(valor, datetime):
        return valor.date()
    if isinstance(valor, date):
        return valor
    return para_datahora(valor).date()
```

```python
"""Conversão de valores de domínio para JSON (auditoria, config)."""

from collections.abc import Mapping
from datetime import date, datetime
from decimal import Decimal
from enum import Enum
from typing import Any


def para_json(valor: Any) -> Any:
    if valor is None or isinstance(valor, bool | int | str):
        return valor
    if isinstance(valor, Enum):
        return valor.value
    if isinstance(valor, Decimal):
        return str(valor)
    if isinstance(valor, datetime | date):
        return valor.isoformat()
    if isinstance(valor, float):
        return str(Decimal(str(valor)))
    if isinstance(valor, Mapping):
        return {str(k): para_json(v) for k, v in valor.items()}
    if isinstance(valor, set | frozenset):
        return sorted(para_json(v) for v in valor)
    if isinstance(valor, list | tuple):
        return [para_json(v) for v in valor]
    raise TypeError(f"Tipo não serializável: {type(valor).__name__}")
```

```python
"""Infra de banco: Base declarativa, engine SQLite, fábrica de sessões e transação."""

from collections.abc import Iterator
from contextlib import contextmanager
from datetime import datetime
from typing import Any

from sqlalchemy import Engine, MetaData, create_engine, event
from sqlalchemy.orm import DeclarativeBase, Mapped, Session, mapped_column, sessionmaker

from cfauto.core import clock

CONVENCAO_NOMES = {
    "ix": "ix_%(column_0_label)s",
    "uq": "uq_%(table_name)s_%(column_0_name)s",
    "ck": "ck_%(table_name)s_%(constraint_name)s",
    "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s",
    "pk": "pk_%(table_name)s",
}


class Base(DeclarativeBase):
    metadata = MetaData(naming_convention=CONVENCAO_NOMES)


def _agora() -> datetime:
    return clock.agora()


class ComTimestamps:
    criado_em: Mapped[datetime] = mapped_column(default=_agora)
    atualizado_em: Mapped[datetime] = mapped_column(default=_agora, onupdate=_agora)


def criar_engine(url: str) -> Engine:
    engine = create_engine(url)
    em_memoria = ":memory:" in url

    @event.listens_for(engine, "connect")
    def _pragmas(conexao: Any, _registro: Any) -> None:
        cursor = conexao.cursor()
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.execute("PRAGMA busy_timeout=5000")
        if not em_memoria:
            cursor.execute("PRAGMA journal_mode=WAL")
        cursor.close()

    return engine


def criar_fabrica(engine: Engine) -> sessionmaker[Session]:
    return sessionmaker(engine, expire_on_commit=False)


@contextmanager
def transacao(fabrica: sessionmaker[Session]) -> Iterator[Session]:
    sessao = fabrica()
    try:
        yield sessao
        sessao.commit()
    except BaseException:
        sessao.rollback()
        raise
    finally:
        sessao.close()
```

```python
"""Decimal armazenado como inteiro escalado: soma exata no SQLite."""

from decimal import ROUND_HALF_UP, Decimal
from typing import Any

from sqlalchemy import Dialect, Integer
from sqlalchemy.types import TypeDecorator


class DecimalEscalado(TypeDecorator[Decimal]):
    impl = Integer
    cache_ok = True

    def __init__(self, casas: int) -> None:
        super().__init__()
        self.casas = casas
        self._fator = Decimal(10) ** casas
        self._quantum = Decimal(1).scaleb(-casas)

    def process_bind_param(self, value: Any, dialect: Dialect) -> int | None:
        if value is None:
            return None
        escalado = Decimal(value) * self._fator
        return int(escalado.to_integral_value(rounding=ROUND_HALF_UP))

    def process_result_value(self, value: Any, dialect: Dialect) -> Decimal | None:
        if value is None:
            return None
        return (Decimal(value) / self._fator).quantize(self._quantum)


def Dinheiro() -> DecimalEscalado:  # noqa: N802 - usado como tipo de coluna
    return DecimalEscalado(2)


def Quantidade() -> DecimalEscalado:  # noqa: N802 - usado como tipo de coluna
    return DecimalEscalado(3)
```

```python
"""Execução programática do Alembic (funciona também empacotado pelo PyInstaller)."""

from pathlib import Path

from alembic import command
from alembic.config import Config
from alembic.runtime.migration import MigrationContext
from alembic.script import ScriptDirectory
from sqlalchemy import Engine

PASTA_MIGRACOES = Path(__file__).parent.parent / "migrations"


def _config() -> Config:
    cfg = Config()
    cfg.set_main_option("script_location", str(PASTA_MIGRACOES))
    return cfg


def migrar(engine: Engine) -> None:
    cfg = _config()
    with engine.begin() as conexao:
        cfg.attributes["connection"] = conexao
        command.upgrade(cfg, "head")


def migracoes_pendentes(engine: Engine) -> bool:
    heads = set(ScriptDirectory.from_config(_config()).get_heads())
    with engine.connect() as conexao:
        atuais = set(MigrationContext.configure(conexao).get_current_heads())
    return atuais != heads
```

```python
import hashlib
import hmac

from argon2 import PasswordHasher
from argon2.exceptions import InvalidHashError, VerificationError

_hasher = PasswordHasher()


def gerar_hash(senha: str) -> str:
    return _hasher.hash(senha)


def verificar(hash_: str, senha: str) -> bool:
    try:
        return _hasher.verify(hash_, senha)
    except (VerificationError, InvalidHashError):
        return False


def precisa_rehash(hash_: str) -> bool:
    return _hasher.check_needs_rehash(hash_)


def verificar_legado(hash_hex: str, senha: str) -> bool:
    """SHA-256 hex sem sal, usado pela planilha USUÁRIOS do sistema original."""
    calculado = hashlib.sha256(senha.encode("utf-8")).hexdigest()
    return hmac.compare_digest(calculado, hash_hex.strip().lower())
```

```python
from dataclasses import dataclass, field
from datetime import timedelta
from typing import Any

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from cfauto.auth import senha as senhas
from cfauto.auth.models import Usuario
from cfauto.core import clock
from cfauto.core.audit import registrar
from cfauto.core.errors import (
    CredenciaisInvalidas,
    LoginDuplicado,
    PermissaoNegada,
    RegistroNaoEncontrado,
    UsuarioBloqueado,
    ValorInvalido,
)
from cfauto.core.permissoes import Ator, Modulo, exigir_admin

MAX_TENTATIVAS = 5
TEMPO_BLOQUEIO = timedelta(minutes=5)
SENHA_MINIMA = 6


@dataclass(frozen=True)
class NovoUsuario:
    nome: str
    login: str
    senha: str
    admin: bool = False
    modulos: frozenset[Modulo] = field(default_factory=frozenset)
    funcao: str = ""


@dataclass(frozen=True)
class AlteracaoUsuario:
    nome: str
    funcao: str
    admin: bool
    modulos: frozenset[Modulo]
    ativo: bool


def para_ator(u: Usuario) -> Ator:
    return Ator(
        id=u.id,
        nome=u.nome,
        login=u.login,
        admin=u.admin,
        modulos=frozenset(Modulo(m) for m in u.modulos),
    )


def _snapshot(u: Usuario) -> dict[str, Any]:
    return {
        "nome": u.nome,
        "funcao": u.funcao,
        "login": u.login,
        "admin": u.admin,
        "modulos": sorted(u.modulos),
        "ativo": u.ativo,
    }


def _normalizar_login(login: str) -> str:
    normalizado = login.strip().lower()
    if not normalizado:
        raise ValorInvalido("Login obrigatório.")
    return normalizado


def _validar_senha(senha: str) -> None:
    if len(senha) < SENHA_MINIMA:
        raise ValorInvalido(f"A senha deve ter ao menos {SENHA_MINIMA} caracteres.")


def _obter(s: Session, usuario_id: int) -> Usuario:
    u = s.get(Usuario, usuario_id)
    if u is None:
        raise RegistroNaoEncontrado("Usuário não encontrado.")
    return u


def existe_usuario(s: Session) -> bool:
    return (s.scalar(select(func.count()).select_from(Usuario)) or 0) > 0


def criar_usuario(s: Session, ator: Ator | None, dados: NovoUsuario) -> Ator:
    if ator is None:
        if existe_usuario(s):
            raise PermissaoNegada("Já existem usuários; entre como administrador.")
        if not dados.admin:
            raise ValorInvalido("O primeiro usuário deve ser administrador.")
    else:
        exigir_admin(ator)
    nome = dados.nome.strip()
    if not nome:
        raise ValorInvalido("Nome obrigatório.")
    login = _normalizar_login(dados.login)
    _validar_senha(dados.senha)
    if s.scalar(select(Usuario.id).where(Usuario.login == login)) is not None:
        raise LoginDuplicado(f"O login '{login}' já existe.")
    u = Usuario(
        nome=nome,
        funcao=dados.funcao.strip(),
        login=login,
        senha_hash=senhas.gerar_hash(dados.senha),
        admin=dados.admin,
        modulos=sorted(m.value for m in dados.modulos),
    )
    s.add(u)
    s.flush()
    registrar(s, ator.id if ator else u.id, "criar", "usuario", u.id, None, _snapshot(u))
    return para_ator(u)


def autenticar(s: Session, login: str, senha: str) -> Ator:
    u = s.scalar(select(Usuario).where(Usuario.login == _normalizar_login(login)))
    if u is None or not u.ativo:
        raise CredenciaisInvalidas()
    agora = clock.agora()
    if u.bloqueado_ate is not None and u.bloqueado_ate > agora:
        raise UsuarioBloqueado(u.bloqueado_ate)
    if u.senha_hash:
        ok = senhas.verificar(u.senha_hash, senha)
        if ok and senhas.precisa_rehash(u.senha_hash):
            u.senha_hash = senhas.gerar_hash(senha)
    elif u.hash_legado:
        ok = senhas.verificar_legado(u.hash_legado, senha)
        if ok:
            u.senha_hash = senhas.gerar_hash(senha)
            u.hash_legado = None
    else:
        ok = False
    if not ok:
        u.tentativas_falhas += 1
        if u.tentativas_falhas >= MAX_TENTATIVAS:
            u.bloqueado_ate = agora + TEMPO_BLOQUEIO
            u.tentativas_falhas = 0
        s.commit()  # a falha precisa persistir mesmo com a exceção abaixo
        raise CredenciaisInvalidas()
    u.tentativas_falhas = 0
    u.bloqueado_ate = None
    s.flush()
    return para_ator(u)


def atualizar_usuario(s: Session, ator: Ator, usuario_id: int, dados: AlteracaoUsuario) -> Ator:
    exigir_admin(ator)
    u = _obter(s, usuario_id)
    if u.admin and u.ativo and (not dados.admin or not dados.ativo):
        outros = s.scalar(
            select(func.count())
            .select_from(Usuario)
            .where(Usuario.admin.is_(True), Usuario.ativo.is_(True), Usuario.id != u.id)
        )
        if outros == 0:
            raise ValorInvalido("Deve existir ao menos um administrador ativo.")
    nome = dados.nome.strip()
    if not nome:
        raise ValorInvalido("Nome obrigatório.")
    antes = _snapshot(u)
    u.nome = nome
    u.funcao = dados.funcao.strip()
    u.admin = dados.admin
    u.modulos = sorted(m.value for m in dados.modulos)
    u.ativo = dados.ativo
    s.flush()
    registrar(s, ator.id, "alterar", "usuario", u.id, antes, _snapshot(u))
    return para_ator(u)
```

```python
"""Chave de licença: base64url(payload_json) + '.' + base64url(assinatura Ed25519)."""

import base64
import binascii
import json
from dataclasses import dataclass
from datetime import date

from cryptography.exceptions import InvalidSignature
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.ed25519 import (
    Ed25519PrivateKey,
    Ed25519PublicKey,
)

from cfauto.core.errors import LicencaInvalida


@dataclass(frozen=True)
class PayloadLicenca:
    cliente: str
    emitida: date
    expira: date
    maquina: str | None = None


def _b64e(dados: bytes) -> str:
    return base64.urlsafe_b64encode(dados).rstrip(b"=").decode("ascii")


def _b64d(texto: str) -> bytes:
    return base64.urlsafe_b64decode(texto + "=" * (-len(texto) % 4))


def _corpo(payload: PayloadLicenca) -> bytes:
    dados = {
        "cliente": payload.cliente,
        "emitida": payload.emitida.isoformat(),
        "expira": payload.expira.isoformat(),
        "maquina": payload.maquina,
    }
    return json.dumps(dados, sort_keys=True, separators=(",", ":")).encode("utf-8")


def assinar(payload: PayloadLicenca, privada: Ed25519PrivateKey) -> str:
    corpo = _corpo(payload)
    return f"{_b64e(corpo)}.{_b64e(privada.sign(corpo))}"


def verificar_assinatura(chave: str, publica: Ed25519PublicKey) -> PayloadLicenca:
    try:
        corpo_b64, assinatura_b64 = chave.strip().split(".")
        corpo = _b64d(corpo_b64)
        publica.verify(_b64d(assinatura_b64), corpo)
        dados = json.loads(corpo)
        return PayloadLicenca(
            cliente=str(dados["cliente"]),
            emitida=date.fromisoformat(dados["emitida"]),
            expira=date.fromisoformat(dados["expira"]),
            maquina=dados.get("maquina"),
        )
    except (ValueError, KeyError, TypeError, binascii.Error, InvalidSignature) as erro:
        raise LicencaInvalida("Licença inválida.") from erro


def carregar_publica_hex(hex_: str) -> Ed25519PublicKey:
    return Ed25519PublicKey.from_public_bytes(bytes.fromhex(hex_))


def publica_para_hex(publica: Ed25519PublicKey) -> str:
    return publica.public_bytes(
        encoding=serialization.Encoding.Raw, format=serialization.PublicFormat.Raw
    ).hex()
```

```python
import platform

def _serial_volume() -> str:
    if sys.platform != "win32":
        return ""
    import ctypes

    serial = ctypes.c_ulong(0)
    ok = ctypes.windll.kernel32.GetVolumeInformationW(  # type: ignore[attr-defined]
        ctypes.c_wchar_p("C:\\"), None, 0, ctypes.byref(serial), None, None, None, 0
    )
    return str(serial.value) if ok else ""


def id_maquina() -> str:
    base = f"{platform.node()}|{_serial_volume()}"
    return hashlib.sha256(base.encode("utf-8")).hexdigest()[:16]
```

```python
import argparse
from datetime import date, timedelta
from pathlib import Path

from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey

from cfauto.licenca.formato import PayloadLicenca, assinar, publica_para_hex
from cfauto.licenca.maquina import id_maquina

PRIVADA_PADRAO = Path.home() / ".cfauto-licenca" / "privada.pem"


def gerar_par(caminho_privada: Path) -> str:
    if caminho_privada.exists():
        raise FileExistsError(f"{caminho_privada} já existe; não vou sobrescrever.")
    caminho_privada.parent.mkdir(parents=True, exist_ok=True)
    privada = Ed25519PrivateKey.generate()
    caminho_privada.write_bytes(
        privada.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=serialization.NoEncryption(),
        )
    )
    return publica_para_hex(privada.public_key())


def _carregar_privada(caminho: Path) -> Ed25519PrivateKey:
    chave = serialization.load_pem_private_key(caminho.read_bytes(), password=None)
    if not isinstance(chave, Ed25519PrivateKey):
        raise ValueError("A chave privada não é Ed25519.")
    return chave


def emitir_chave(
    caminho_privada: Path, cliente: str, expira: date, maquina: str | None = None
) -> str:
    payload = PayloadLicenca(cliente=cliente, emitida=date.today(), expira=expira, maquina=maquina)
    return assinar(payload, _carregar_privada(caminho_privada))

    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="comando", required=True)

    p_gerar = sub.add_parser("gerar-chaves")

    p_emitir.add_argument("--cliente", required=True)
    grupo = p_emitir.add_mutually_exclusive_group(required=True)
    grupo.add_argument("--dias", type=int)
    grupo.add_argument("--expira", type=date.fromisoformat)
    p_emitir.add_argument("--maquina")

    sub.add_parser("id-maquina")

    args = parser.parse_args()
    if args.comando == "gerar-chaves":
        print(gerar_par(args.saida))
    elif args.comando == "emitir":
        expira = args.expira or date.today() + timedelta(days=args.dias)
        print(emitir_chave(args.privada, args.cliente, expira, args.maquina))
    else:
        print(id_maquina())


if __name__ == "__main__":
    main()
```

```python
from dataclasses import dataclass
from datetime import timedelta
from enum import StrEnum

from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PublicKey
from sqlalchemy.orm import Session

from cfauto.core import clock
from cfauto.core.audit import registrar
from cfauto.core.errors import LicencaExpirada, LicencaInvalida
from cfauto.licenca.chave_publica import CHAVE_PUBLICA_HEX
from cfauto.licenca.formato import PayloadLicenca, carregar_publica_hex, verificar_assinatura
from cfauto.licenca.maquina import id_maquina
from cfauto.licenca.models import LicencaEstado

TOLERANCIA_RELOGIO = timedelta(days=1)


class StatusLicenca(StrEnum):
    VALIDA = "VALIDA"
    AUSENTE = "AUSENTE"
    INVALIDA = "INVALIDA"
    EXPIRADA = "EXPIRADA"
    RELOGIO = "RELOGIO"


@dataclass(frozen=True)
class EstadoLicenca:
    status: StatusLicenca
    payload: PayloadLicenca | None = None
    dias_restantes: int | None = None

    @property
    def valida(self) -> bool:
        return self.status is StatusLicenca.VALIDA

    @property
    def mensagem(self) -> str:
        if self.status is StatusLicenca.VALIDA and self.payload:
            return f"Licença válida até {self.payload.expira:%d/%m/%Y}."
        if self.status is StatusLicenca.EXPIRADA and self.payload:
            return f"A licença expirou em {self.payload.expira:%d/%m/%Y}."
        if self.status is StatusLicenca.RELOGIO:
            return (
                "Relógio do sistema inconsistente. Corrija a data do Windows e abra o CFAUTO "
                "novamente."
            )
        if self.status is StatusLicenca.INVALIDA:
            return "Licença inválida."
        return "Nenhuma licença instalada."


def _publica(publica: Ed25519PublicKey | None) -> Ed25519PublicKey:
    return publica or carregar_publica_hex(CHAVE_PUBLICA_HEX)


def _conferir_maquina(payload: PayloadLicenca, maquina_atual: str | None) -> None:
    if payload.maquina and payload.maquina != (maquina_atual or id_maquina()):
        raise LicencaInvalida("Esta licença foi emitida para outra máquina.")


def _linha(s: Session) -> LicencaEstado:
    linha = s.get(LicencaEstado, 1)
    if linha is None:
        linha = LicencaEstado(id=1)
        s.add(linha)
        s.flush()
    return linha


def validar(
    chave: str, *, publica: Ed25519PublicKey | None = None, maquina_atual: str | None = None
) -> PayloadLicenca:
    payload = verificar_assinatura(chave, _publica(publica))
    _conferir_maquina(payload, maquina_atual)
    if payload.expira < clock.hoje():
        raise LicencaExpirada(f"A licença expirou em {payload.expira:%d/%m/%Y}.")
    return payload


def instalar(
    s: Session,
    chave: str,
    *,
    publica: Ed25519PublicKey | None = None,
    maquina_atual: str | None = None,
) -> EstadoLicenca:
    payload = validar(chave, publica=publica, maquina_atual=maquina_atual)
    linha = _linha(s)
    linha.chave = chave.strip()
    s.flush()
    registrar(
        s, None, "instalar", "licenca", payload.cliente, None,
        {"cliente": payload.cliente, "expira": payload.expira},
    )
    return avaliar(s, publica=publica, maquina_atual=maquina_atual)


def avaliar(
    s: Session, *, publica: Ed25519PublicKey | None = None, maquina_atual: str | None = None
) -> EstadoLicenca:
    linha = _linha(s)
    if not linha.chave:
        return EstadoLicenca(StatusLicenca.AUSENTE)
    hoje = clock.hoje()
    if linha.ultima_data_vista and hoje < linha.ultima_data_vista - TOLERANCIA_RELOGIO:
        return EstadoLicenca(StatusLicenca.RELOGIO)
    try:
        payload = verificar_assinatura(linha.chave, _publica(publica))
        _conferir_maquina(payload, maquina_atual)
    except LicencaInvalida:
```

```python
def configurar_logs(pasta: Path, nivel: int = logging.INFO) -> None:
    pasta.mkdir(parents=True, exist_ok=True)
    logger = logging.getLogger("cfauto")
    logger.setLevel(nivel)
    arquivo = str(pasta / "cfauto.log")
```

```python
"""Criação/migração do banco local e backups."""

import logging
import os
import sqlite3
from dataclasses import dataclass
from pathlib import Path

from sqlalchemy import Engine
from sqlalchemy.orm import Session, sessionmaker

from cfauto.auth.service import existe_usuario
from cfauto.core import clock
from cfauto.core.db import criar_engine, criar_fabrica, transacao
from cfauto.core.migracoes import migracoes_pendentes, migrar

log = logging.getLogger("cfauto.sistema")


@dataclass(frozen=True)
class Caminhos:
    base: Path

    @property
    def db(self) -> Path:
        return self.base / "cfauto.db"

    @property
    def backups(self) -> Path:
        return self.base / "backups"

    @property
    def logs(self) -> Path:
        return self.base / "logs"


def caminhos_padrao() -> Caminhos:
    home = os.environ.get("CFAUTO_HOME")
    if home:
        return Caminhos(Path(home))
    appdata = os.environ.get("APPDATA")
    return Caminhos((Path(appdata) if appdata else Path.home()) / "CFAUTO")


def fazer_backup(db: Path, destino: Path, manter: int = 7) -> Path | None:
    if not db.exists():
        return None
    destino.mkdir(parents=True, exist_ok=True)
    alvo = destino / f"cfauto-{clock.agora():%Y%m%d-%H%M%S}.db"
    contador = 1
    while alvo.exists():
        alvo = destino / f"cfauto-{clock.agora():%Y%m%d-%H%M%S}-{contador}.db"
        contador += 1
    origem = sqlite3.connect(db)
    copia = sqlite3.connect(alvo)
    try:
        origem.backup(copia)
    finally:
        copia.close()
        origem.close()
    for antigo in sorted(destino.glob("cfauto-*.db"))[:-manter]:
        antigo.unlink()
    log.info("backup criado em %s", alvo)
    return alvo


@dataclass
class Ambiente:
    caminhos: Caminhos
    engine: Engine
    fabrica: sessionmaker[Session]
    primeira_execucao: bool


def inicializar(caminhos: Caminhos) -> Ambiente:
    caminhos.base.mkdir(parents=True, exist_ok=True)
    existia = caminhos.db.exists()
    engine = criar_engine(f"sqlite:///{caminhos.db}")
    if existia and migracoes_pendentes(engine):
        fazer_backup(caminhos.db, caminhos.backups)
    migrar(engine)
    fabrica = criar_fabrica(engine)
    with transacao(fabrica) as s:
        primeira = not existe_usuario(s)
    return Ambiente(caminhos, engine, fabrica, primeira)
```

```python
from typing import Any

RegraCatchup = Callable[[datetime | None, datetime], bool]


@dataclass(frozen=True)
class Job:
    nome: str
    executar: Callable[[], None]
    cron: dict[str, Any]  # argumentos do CronTrigger do APScheduler

        return ultima is None or agora - ultima >= intervalo
```

```python
import logging
from collections.abc import Iterable
from datetime import datetime

from sqlalchemy.orm import Session, sessionmaker

from cfauto.core import clock, config
from cfauto.core.db import transacao
from cfauto.jobs.modelo import Job

log = logging.getLogger("cfauto.jobs")


def _chave(nome: str) -> str:
    return f"job.{nome}.ultima_execucao"


def ultima_execucao(fabrica: sessionmaker[Session], nome: str) -> datetime | None:
    with transacao(fabrica) as s:
        valor = config.obter(s, _chave(nome))
    return None if valor is None else datetime.fromisoformat(valor)


def rodar(fabrica: sessionmaker[Session], job: Job) -> None:
    log.info("job %s iniciado", job.nome)
    try:
        job.executar()
    except Exception:
        log.exception("job %s falhou", job.nome)
        raise
    with transacao(fabrica) as s:
        config.definir(s, _chave(job.nome), clock.agora())
    log.info("job %s concluído", job.nome)


def catchup(fabrica: sessionmaker[Session], jobs: Iterable[Job]) -> dict[str, Exception | None]:
    resultado: dict[str, Exception | None] = {}
    agora = clock.agora()
    for job in jobs:
        if not job.precisa_catchup(ultima_execucao(fabrica, job.nome), agora):
            continue
        try:
            rodar(fabrica, job)
            resultado[job.nome] = None
        except Exception as erro:  # já registrado em log por rodar()
            resultado[job.nome] = erro
    return resultado
```

`cfauto/jobs/agendador.py`:
```python
from collections.abc import Sequence
from contextlib import suppress

from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from sqlalchemy.orm import Session, sessionmaker

from cfauto.jobs.modelo import Job
from cfauto.jobs.runner import rodar


class Agendador:
    def __init__(self, fabrica: sessionmaker[Session], jobs: Sequence[Job]) -> None:
        self._fabrica = fabrica
        self._jobs = list(jobs)
        self._scheduler = BackgroundScheduler()

    def _executar(self, job: Job) -> None:
        # rodar() registra a falha no log; a thread do agendador não pode morrer.
        with suppress(Exception):
            rodar(self._fabrica, job)

    def iniciar(self) -> None:
        for job in self._jobs:
            self._scheduler.add_job(
                self._executar,
                CronTrigger(**job.cron),
                args=[job],
                id=job.nome,
                coalesce=True,
                max_instances=1,
                misfire_grace_time=3600,
            )
        self._scheduler.start()

    def nomes_agendados(self) -> list[str]:
        return [j.id for j in self._scheduler.get_jobs()]

    def parar(self) -> None:
        if self._scheduler.running:
            self._scheduler.shutdown(wait=False)
```

```python
from PySide6.QtCore import QEvent, QObject, QTimer, Signal
from PySide6.QtWidgets import QApplication

_EVENTOS = {
    QEvent.Type.MouseMove,
    QEvent.Type.MouseButtonPress,
    QEvent.Type.KeyPress,
    QEvent.Type.Wheel,
}


class MonitorInatividade(QObject):
    inativo = Signal()

    def __init__(self, intervalo_ms: int, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._timer = QTimer(self)
        self._timer.setSingleShot(True)
        self._timer.setInterval(intervalo_ms)
        self._timer.timeout.connect(self.inativo)
        app = QApplication.instance()
        if app is not None:
            app.installEventFilter(self)

    def eventFilter(self, obj: QObject, evento: QEvent) -> bool:  # noqa: N802 - API do Qt
        if evento.type() in _EVENTOS and self._timer.isActive():
            self._timer.start()
        return False

    def reiniciar(self) -> None:
        self._timer.start()

    def parar(self) -> None:
        self._timer.stop()
```

```python
        qapp.aboutToQuit.connect(self._ao_sair)

    def executar(self) -> int:
        if self._ambiente.primeira_execucao:
            assistente = AssistentePrimeiraExecucao(self._fabrica)
            if not assistente.exec():  # exec() devolve int: 1 = aceito
                return 0
        if not self._garantir_licenca():
            return 0
        falhas = {n: e for n, e in catchup(self._fabrica, self._jobs).items() if e is not None}
        if falhas:
            QMessageBox.warning(
                None, "CFAUTO", "Tarefas automáticas com falha: " + ", ".join(sorted(falhas))
            )
        self._agendador.iniciar()
        if not self._login():
            return 0
        return self._qapp.exec()

    def _garantir_licenca(self) -> bool:
        while True:
            with transacao(self._fabrica) as s:
                estado = avaliar(s)
            if estado.valida:
                self._estado_licenca = estado
                return True
            dialogo = DialogoLicenca(self._fabrica, estado)
            if not dialogo.exec():
                return False

    def _login(self) -> bool:
        dialogo = DialogoLogin(self._fabrica)
        if not dialogo.exec() or dialogo.ator is None:
            return False
        self._abrir_janela(dialogo.ator)
        return True

    def _abrir_janela(self, ator: Ator) -> None:
        assert self._estado_licenca is not None
        self._ator = ator
        self._janela = JanelaPrincipal(self._fabrica, ator, self._estado_licenca)
        self._janela.sair.connect(self._logout)
        self._janela.encerrar.connect(self._qapp.quit)
        self._janela.show()
        self._monitor.reiniciar()

    def _logout(self) -> None:
        self._monitor.parar()
        if self._janela is not None:
            self._janela.close()
            self._janela.deleteLater()
            self._janela = None
        if not self._login():
            self._qapp.quit()

    def _bloquear(self) -> None:
        if self._janela is None or self._ator is None:
            return
        self._janela.hide()
        dialogo = DialogoLogin(self._fabrica, login_fixo=self._ator.login)
        if dialogo.exec():
            self._janela.show()
            self._monitor.reiniciar()
        else:
            self._logout()

    def _ao_sair(self) -> None:
        self._agendador.parar()
        try:
            fazer_backup(self._ambiente.caminhos.db, self._ambiente.caminhos.backups)
        except Exception:
            log.exception("backup ao sair falhou")


def main(argv: list[str] | None = None) -> int:
    argumentos = sys.argv[1:] if argv is None else argv
    caminhos = caminhos_padrao()
    configurar_logs(caminhos.logs)
    if "--verificar-instalacao" in argumentos:
        ambiente = inicializar(caminhos)
        ambiente.engine.dispose()
        return 0
    qapp = QApplication(sys.argv)
    qapp.setApplicationName("CFAUTO")
    qapp.setQuitOnLastWindowClosed(False)
    try:
        ambiente = inicializar(caminhos)
    except Exception:
        log.exception("falha ao inicializar o banco")
        QMessageBox.critical(
            None, "CFAUTO", f"Falha ao iniciar o banco de dados. Veja o log em {caminhos.logs}."
        )
        return 1
    return Aplicacao(qapp, ambiente).executar()
```

```python
class TelaUsuarios(QWidget):
    def __init__(self, fabrica: sessionmaker[Session], ator: Ator) -> None:
        super().__init__()
        self._fabrica = fabrica
        self._ator = ator
        self._usuarios: list[Usuario] = []

        self.tabela = QTableWidget(0, len(_COLUNAS))
        self.tabela.setHorizontalHeaderLabels(_COLUNAS)
        self.tabela.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
        self.tabela.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.tabela.horizontalHeader().setStretchLastSection(True)

        novo = QPushButton("Novo")
        novo.clicked.connect(self._novo)
        editar = QPushButton("Editar")
        editar.clicked.connect(self._editar)
        senha = QPushButton("Alterar senha")
        senha.clicked.connect(self._alterar_senha)
        botoes = QHBoxLayout()
        for botao in (novo, editar, senha):
            botoes.addWidget(botao)
        botoes.addStretch()

        layout = QVBoxLayout(self)
        layout.addLayout(botoes)
        layout.addWidget(self.tabela)
        self.recarregar()

    def recarregar(self) -> None:
        with transacao(self._fabrica) as s:
            self._usuarios = listar_usuarios(s, self._ator)
        self.tabela.setRowCount(len(self._usuarios))
        for linha, u in enumerate(self._usuarios):
            modulos = "Todos" if u.admin else ", ".join(TITULOS[Modulo(m)] for m in u.modulos)
            valores = [u.nome, u.login, u.funcao, "Sim" if u.admin else "Não", modulos,
                       "Sim" if u.ativo else "Não"]
            for coluna, valor in enumerate(valores):
                self.tabela.setItem(linha, coluna, QTableWidgetItem(valor))

    def ids_na_tabela(self) -> list[int]:
        return [u.id for u in self._usuarios]

    def usuario_da_linha(self, linha: int) -> Usuario:
        return self._usuarios[linha]

    def _selecionado(self) -> Usuario | None:
        linha = self.tabela.currentRow()
        return None if linha < 0 else self._usuarios[linha]

    def _novo(self) -> None:
        if DialogoUsuario(self._fabrica, self._ator, parent=self).exec():
            self.recarregar()

    def _editar(self) -> None:
        usuario = self._selecionado()
        if usuario and DialogoUsuario(self._fabrica, self._ator, usuario, self).exec():
            self.recarregar()

    def _alterar_senha(self) -> None:
        usuario = self._selecionado()
        if usuario is None:
            return
        nova, ok = QInputDialog.getText(
            self, "Alterar senha", f"Nova senha de {usuario.nome}:", QLineEdit.EchoMode.Password
        )
        if not ok:
            return
        try:
            with transacao(self._fabrica) as s:
                alterar_senha(s, self._ator, usuario.id, nova)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
```

```python
def _json(valor: object) -> str:

class TelaAuditoria(QWidget):

        super().__init__()
        self._fabrica = fabrica
        self._ator = ator

        botao.clicked.connect(self.filtrar)

        filtros = QHBoxLayout()
        filtros.addWidget(QLabel("Entidade"))
        filtros.addWidget(self.filtro_entidade)
        filtros.addWidget(QLabel("De"))
        filtros.addWidget(self.desde)
        filtros.addWidget(QLabel("Até"))
        filtros.addWidget(self.ate)
        filtros.addWidget(botao)

        self.tabela = QTableWidget(0, len(_COLUNAS))
        self.tabela.setHorizontalHeaderLabels(_COLUNAS)
        self.tabela.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.tabela.horizontalHeader().setStretchLastSection(True)

        layout = QVBoxLayout(self)
        layout.addLayout(filtros)
        layout.addWidget(self.tabela)
        self.filtrar()

    def filtrar(self) -> None:
        desde = datetime.combine(self.desde.date().toPython(), time.min)
        ate = datetime.combine(self.ate.date().toPython(), time.max)
        with transacao(self._fabrica) as s:
            registros = consultar(
                s, self._ator, entidade=self.filtro_entidade.text().strip() or None,
                desde=desde, ate=ate,
            )
        self.tabela.setRowCount(len(registros))
        for linha, r in enumerate(registros):
            valores = [
                f"{r.timestamp:%d/%m/%Y %H:%M:%S}", r.usuario_nome or "(sistema)", r.acao,
                r.entidade, r.entidade_id or "", _json(r.antes), _json(r.depois),
            ]
            for coluna, valor in enumerate(valores):
                self.tabela.setItem(linha, coluna, QTableWidgetItem(valor))
```
