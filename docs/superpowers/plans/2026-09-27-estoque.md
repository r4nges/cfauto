**Goal:** Entregar o módulo Estoque (produto, serviço, ajuste com motivo, livro de movimentos, entrada por XML de NF-e com conferência, estorno) e o cadastro de Bens Físicos em Configurações › CFAUTO.

## Global Constraints

- Idioma do código de domínio, mensagens e UI: português (identificadores sem acento: `produto`, `movimentar`). Mensagens ao usuário com acentuação correta.
- Python 3.13 via uv; rode tudo na raiz `C:\Users\Admin\Documents\CFAUTO` com `uv run ...`. O hook RTK esconde o resumo do pytest: use `rtk proxy uv run pytest ...` para ver a saída real.
- Dinheiro, custo e quantidade: `Decimal`, nunca `float`. Colunas: `Dinheiro()` (2 casas), `Custo()` (4 casas), `Quantidade()` (3 casas), todos inteiros escalados (`cfauto.core.tipos`). Enums no banco via `EnumTexto(Enum)` (VARCHAR com o valor do `StrEnum`).
- Data atual só por `cfauto.core.clock.agora()` / `clock.hoje()`.
- Datas externas passam por `cfauto.core.dates` na borda. Números do XML da NF-e: `Decimal(texto)` direto — **nunca** `para_decimal` (o XML usa ponto decimal; `"1.000"` é 1).
- Acesso a dados sempre por nome de atributo; proibido indexar tuplas/linhas posicionais para campos de negócio (use `.label()` em consultas com colunas soltas).
- `ui/` nunca monta consulta/SQL; chama funções de serviço e pode receber objetos ORM retornados por elas (ruling R2 da Fundação). Relacionamentos lidos pela UI depois da transação são `lazy="selectin"`.
- UI: valores digitados em `QLineEdit` e convertidos com `cfauto.ui.comum.ler_decimal` (formato brasileiro); nunca `QDoubleSpinBox`. Combos guardam o enum como dado e a leitura reconverte (`Destino(combo.currentData())`), porque o Qt devolve `str`.
- Permissões: tudo do Estoque `exigir_modulo(ator, Modulo.ESTOQUE)`; estorno de nota `exigir_admin(ator)`; Bens Físicos `exigir_modulo(ator, Modulo.CFAUTO)` (só admin).
- Nenhuma exceção é engolida devolvendo valor padrão; erros de regra são subclasses de `cfauto.core.errors.ErroDominio`. Auditoria via `cfauto.core.audit.registrar` na mesma transação.
- Testes de UI que exercitam erro fazem `monkeypatch` de `mostrar_erro` no módulo da tela (o `QMessageBox` bloqueia o teste offscreen).
- Antes do `uv run ruff check .`, rode `uv run ruff format <arquivos da task>`: algumas linhas do plano passam de 100 colunas e o formatador as quebra (o pre-commit roda os dois).
- Commits: Conventional Commits, terminando com as linhas de attribution do SEU system-reminder. Escreva caracteres invisíveis só por escape (`"\u00a0"`).

```python
"""Tipos de coluna: Decimal como inteiro escalado (soma exata no SQLite) e enum como texto."""

from decimal import ROUND_HALF_UP, Decimal
from enum import StrEnum
from typing import Any

from sqlalchemy import Dialect, Integer, String
from sqlalchemy.types import TypeDecorator


class DecimalEscalado(TypeDecorator[Decimal]):
    impl = Integer
    cache_ok = True

    def __init__(self, casas: int) -> None:
        super().__init__()
        self.casas = casas
        self._fator = Decimal(10) ** casas
        self._quantum = Decimal(1).scaleb(-casas)

    def process_bind_param(self, value: Any, dialect: Dialect) -> int | None:
        if value is None:
            return None
        if isinstance(value, float | bool):
            raise TypeError(f"Use Decimal, não float: {value!r}")
        escalado = Decimal(value) * self._fator
        return int(escalado.to_integral_value(rounding=ROUND_HALF_UP))

    def process_result_value(self, value: Any, dialect: Dialect) -> Decimal | None:
        if value is None:
            return None
        return (Decimal(value) / self._fator).quantize(self._quantum)


class EnumTexto[E: StrEnum](TypeDecorator[E]):
    """StrEnum guardado pelo valor (VARCHAR); volta do banco como o próprio enum."""

    impl = String
    cache_ok = True

    def __init__(self, enum: type[E], tamanho: int = 20) -> None:
        super().__init__(tamanho)
        self.enum = enum

    def process_bind_param(self, value: Any, dialect: Dialect) -> str | None:
        if value is None:
            return None
        return self.enum(value).value

    def process_result_value(self, value: Any, dialect: Dialect) -> E | None:
        if value is None:
            return None
        return self.enum(value)


def Dinheiro() -> DecimalEscalado:  # noqa: N802 - usado como tipo de coluna
    return DecimalEscalado(2)


def Custo() -> DecimalEscalado:  # noqa: N802 - usado como tipo de coluna
    """Custo unitário: 4 casas (R$ 0,035 não pode virar R$ 0,04)."""
    return DecimalEscalado(4)


def Quantidade() -> DecimalEscalado:  # noqa: N802 - usado como tipo de coluna
    return DecimalEscalado(3)
```

```python
class Produto(ComTimestamps, Base):
    __tablename__ = "produto"

    id: Mapped[int] = mapped_column(primary_key=True)
    codigo: Mapped[str] = mapped_column(String(20), unique=True)
    ean: Mapped[str | None] = mapped_column(String(14), unique=True)
    descricao: Mapped[str] = mapped_column(String(200))
    categoria: Mapped[str] = mapped_column(String(60), default="")
    embalagem: Mapped[str] = mapped_column(String(30), default="Unidade")
    # Unidades de estoque por embalagem; preco_entrada é o custo da embalagem.
    fator: Mapped[Decimal] = mapped_column(Quantidade(), default=Decimal(1))
    data_ult_entrada: Mapped[date | None]
    preco_entrada: Mapped[Decimal] = mapped_column(Custo(), default=Decimal(0))
    preco_venda: Mapped[Decimal] = mapped_column(Dinheiro(), default=Decimal(0))
    preco_promocional: Mapped[Decimal | None] = mapped_column(Dinheiro())
    # Saldo; só muda por cfauto.estoque.movimentos.movimentar().
    estoque: Mapped[Decimal] = mapped_column(Quantidade(), default=Decimal(0))
    estoque_minimo: Mapped[Decimal] = mapped_column(Quantidade(), default=Decimal(0))
    estoque_minimo_manual: Mapped[bool] = mapped_column(default=False)
    destino: Mapped[Destino] = mapped_column(EnumTexto(Destino), default=Destino.REVENDA)
    markup_pct: Mapped[Decimal | None] = mapped_column(DecimalEscalado(2))
    markup_manual: Mapped[bool] = mapped_column(default=False)
    ativo: Mapped[bool] = mapped_column(default=True)

    @property
    def preco_fracionado(self) -> Decimal:
        """Custo por unidade de estoque."""
        return self.preco_entrada / self.fator

    @property
    def margem(self) -> Decimal | None:
        """(venda − custo unitário) ÷ venda; None sem preço de venda ou sem custo."""
        if self.preco_venda <= 0 or self.preco_entrada <= 0:
            return None
        return (self.preco_venda - self.preco_fracionado) / self.preco_venda


class Servico(ComTimestamps, Base):
    __tablename__ = "servico"

    id: Mapped[int] = mapped_column(primary_key=True)
    codigo: Mapped[str] = mapped_column(String(20), unique=True)
    nome: Mapped[str] = mapped_column(String(120))
    mao_obra_pct: Mapped[Decimal] = mapped_column(DecimalEscalado(2), default=Decimal(0))
    ativo: Mapped[bool] = mapped_column(default=True)
    materiais: Mapped[list["ServicoMaterial"]] = relationship(
        back_populates="servico",
        cascade="all, delete-orphan",
        order_by="ServicoMaterial.id",
        lazy="selectin",
    )

    @property
    def custo_materiais(self) -> Decimal:
        return sum((m.custo_unit * m.qtd for m in self.materiais), Decimal(0))

    @property
    def preco(self) -> Decimal:
        return arredondar(self.custo_materiais * (1 + self.mao_obra_pct / 100))
```

```python
def para_datahora(valor: date | datetime | str) -> datetime:
    if isinstance(valor, datetime):
        return valor
    if isinstance(valor, date):
        return datetime(valor.year, valor.month, valor.day)
    if isinstance(valor, str):
        texto = valor.strip()
        for formato in _FORMATOS_DATAHORA + _FORMATOS_DATA:
            try:
                return datetime.strptime(texto, formato)
            except ValueError:
                continue
        # ISO 8601 com fuso (XML da NF-e: 2026-09-20T10:15:00-03:00): vale a hora de parede.
        with suppress(ValueError):
            return datetime.fromisoformat(texto).replace(tzinfo=None)
    raise DataInvalida(f"Data inválida: {valor!r}")


def para_data(valor: date | datetime | str) -> date:
    if isinstance(valor, datetime):
        return valor.date()
    if isinstance(valor, date):
        return valor
    return para_datahora(valor).date()
```

```python
MILESIMO = Decimal("0.001")
DECIMO_DE_MILESIMO = Decimal("0.0001")


def arredondar_qtd(valor: Decimal) -> Decimal:
    return valor.quantize(MILESIMO, rounding=ROUND_HALF_UP)


def arredondar_custo(valor: Decimal) -> Decimal:
    return valor.quantize(DECIMO_DE_MILESIMO, rounding=ROUND_HALF_UP)


def formatar_qtd(valor: Decimal) -> str:
    """Quantidade para exibição: 1.500 → '1,5'; 1234 → '1.234'."""
    v = arredondar_qtd(valor)
    sinal = "-" if v < 0 else ""
    inteiro, _, fracao = f"{abs(v):.3f}".partition(".")
    grupos = f"{int(inteiro):,}".replace(",", ".")
    fracao = fracao.rstrip("0")
    return f"{sinal}{grupos},{fracao}" if fracao else f"{sinal}{grupos}"


def formatar_pct(fracao: Decimal) -> str:
    """Fração como porcentagem com 1 casa: 0.125 → '12,5%'."""
    pct = (fracao * 100).quantize(Decimal("0.1"), rounding=ROUND_HALF_UP)
    return f"{pct:.1f}".replace(".", ",") + "%"


def formatar_numero(valor: Decimal, casas_minimas: int = 2) -> str:
    """Decimal para campo editável, sem separador de milhar: 1234.5 → '1234,50'."""
    inteiro, _, fracao = format(valor, "f").partition(".")
    fracao = fracao.rstrip("0").ljust(casas_minimas, "0")
    return f"{inteiro},{fracao}" if fracao else inteiro
```

```python
def normalizar_texto(texto: str) -> str:
    decomposto = unicodedata.normalize("NFKD", texto)
    sem_acento = "".join(c for c in decomposto if not unicodedata.combining(c))
```

```python
"""Única porta de alteração do saldo de estoque: todo movimento fica registrado."""

from decimal import Decimal

from sqlalchemy.orm import Session

from cfauto.core import clock
from cfauto.core.errors import ValorInvalido
from cfauto.core.money import arredondar_qtd
from cfauto.core.permissoes import Ator
from cfauto.estoque.models import (
    MotivoAjuste,
    MovimentoEstoque,
    NotaEntradaItem,
    Produto,
    TipoMovimento,
)


def movimentar(
    s: Session,
    ator: Ator,
    produto: Produto,
    quantidade: Decimal,
    tipo: TipoMovimento,
    *,
    motivo: MotivoAjuste | None = None,
    observacao: str | None = None,
    nota_item: NotaEntradaItem | None = None,
) -> MovimentoEstoque:
    """Soma `quantidade` (com sinal) ao estoque. Não checa permissão nem saldo negativo:
    quem chama decide."""
    quantidade = arredondar_qtd(quantidade)
    if quantidade == 0:
        raise ValorInvalido("Movimento de estoque com quantidade zero.")
    produto.estoque = produto.estoque + quantidade
    movimento = MovimentoEstoque(
        produto_id=produto.id,
        data_hora=clock.agora(),
        tipo=tipo,
        quantidade=quantidade,
        saldo_apos=produto.estoque,
        usuario_id=ator.id,
        motivo=motivo,
        observacao=observacao,
        nota_entrada_item_id=None if nota_item is None else nota_item.id,
    )
    s.add(movimento)
    s.flush()
    return movimento
```

```python
    if not (texto.isascii() and texto.isdigit()) or len(texto) > TAMANHO_MAXIMO_EAN:
        raise ValorInvalido("O EAN deve ter só dígitos (no máximo 14).")
    return texto


def ean_em_uso(s: Session, ean: str) -> Produto | None:
    return s.scalar(select(Produto).where(Produto.ean == ean))


def novo_codigo_produto(s: Session) -> str:
    return proximo_codigo(s, "produto", digitos=4)


def _obter(s: Session, produto_id: int) -> Produto:
    produto = s.get(Produto, produto_id)
    if produto is None:
        raise RegistroNaoEncontrado("Produto não encontrado.")
    return produto


def _validar(dados: DadosProduto) -> str:
    descricao = dados.descricao.strip()
    if not descricao:
        raise ValorInvalido("Descrição obrigatória.")
    if arredondar_qtd(dados.fator) <= 0:
        raise ValorInvalido("O fator deve ser maior que zero.")
    if dados.preco_entrada < 0 or dados.preco_venda < 0:
        raise ValorInvalido("Preços não podem ser negativos.")
    if dados.preco_promocional is not None and dados.preco_promocional <= 0:
        raise ValorInvalido("O preço promocional deve ser maior que zero.")
    if dados.estoque_minimo < 0:
        raise ValorInvalido("O estoque mínimo não pode ser negativo.")
    return descricao


def _checar_ean(s: Session, ean: str | None, produto_id: int | None) -> None:
    if ean is None:
        return
    dono = ean_em_uso(s, ean)
    if dono is not None and dono.id != produto_id:
        raise EanDuplicado(f"O EAN {ean} já pertence ao produto {dono.codigo} — {dono.descricao}.")


def _aplicar(p: Produto, dados: DadosProduto, descricao: str, ean: str | None) -> None:
    p.ean = ean
    p.descricao = descricao
    p.categoria = dados.categoria.strip()
    p.embalagem = dados.embalagem.strip() or "Unidade"
    p.fator = arredondar_qtd(dados.fator)
    p.preco_entrada = arredondar_custo(dados.preco_entrada)
    p.preco_venda = arredondar(dados.preco_venda)
    p.preco_promocional = (
        None if dados.preco_promocional is None else arredondar(dados.preco_promocional)
    )
    p.destino = dados.destino


def obter_produto(s: Session, ator: Ator, produto_id: int) -> Produto:
    exigir_modulo(ator, Modulo.ESTOQUE)
    return _obter(s, produto_id)


def criar_produto(s: Session, ator: Ator, dados: DadosProduto) -> Produto:
    exigir_modulo(ator, Modulo.ESTOQUE)
    descricao = _validar(dados)
    estoque_inicial = arredondar_qtd(dados.estoque_inicial)
    if estoque_inicial < 0:
        raise ValorInvalido("O estoque inicial não pode ser negativo.")
    ean = normalizar_ean(dados.ean)
    _checar_ean(s, ean, None)
    minimo = arredondar_qtd(dados.estoque_minimo)
    p = Produto(
        codigo=novo_codigo_produto(s),
        estoque=Decimal("0.000"),
        estoque_minimo=minimo,
        estoque_minimo_manual=dados.estoque_minimo_manual or minimo > 0,
        # Carimbada no cadastro mesmo com estoque 0 (como no original); nunca na edição.
        data_ult_entrada=clock.hoje(),
        markup_manual=False,
        ativo=True,
    )
    _aplicar(p, dados, descricao, ean)
    s.add(p)
    s.flush()
    if estoque_inicial > 0:
        movimentar(s, ator, p, estoque_inicial, TipoMovimento.CADASTRO)
    registrar(s, ator.id, "criar", "produto", p.id, None, snapshot_produto(p))
    return p


def atualizar_produto(s: Session, ator: Ator, produto_id: int, dados: DadosProduto) -> Produto:
    """Altera o cadastro. Nunca mexe no estoque nem na data da última entrada."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    p = _obter(s, produto_id)
    descricao = _validar(dados)
    ean = normalizar_ean(dados.ean)
    _checar_ean(s, ean, p.id)
    antes = snapshot_produto(p)
    _aplicar(p, dados, descricao, ean)
    minimo = arredondar_qtd(dados.estoque_minimo)

    p.estoque_minimo = minimo
    s.flush()
    registrar(s, ator.id, "alterar", "produto", p.id, antes, snapshot_produto(p))
    return p

def ajustar_estoque(
    s: Session,
    ator: Ator,
    produto_id: int,
    nova_quantidade: Decimal,
    motivo: MotivoAjuste,
    observacao: str = "",
) -> MovimentoEstoque:
    """Leva o estoque à quantidade contada, registrando a diferença com motivo."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    p = _obter(s, produto_id)
    nova = arredondar_qtd(nova_quantidade)
    if nova < 0:
        raise ValorInvalido("A quantidade não pode ser negativa.")
    obs = observacao.strip()
    if motivo is MotivoAjuste.OUTRO and not obs:
        raise ValorInvalido("Informe a observação quando o motivo for 'Outro'.")
    diferenca = nova - p.estoque
    if diferenca == 0:
        raise ValorInvalido("O estoque já está com essa quantidade.")
    antes = p.estoque
    movimento = movimentar(
        s, ator, p, diferenca, TipoMovimento.AJUSTE, motivo=motivo, observacao=obs or None
    )
    registrar(
        s,
        ator.id,
        "ajustar_estoque",
        "produto",
        p.id,
        {"estoque": antes},
        {"estoque": p.estoque, "motivo": motivo, "observacao": obs or None},
    )
    return movimento


def _casa_codigo(p: Produto, chave: str) -> bool:
    if p.ean is not None and p.ean == chave:
        return True
    if p.codigo == chave:
        return True
    return chave.isdigit() and p.codigo.isdigit() and int(p.codigo) == int(chave)


def buscar_produtos(
    s: Session, ator: Ator, termo: str = "", *, incluir_inativos: bool = False
) -> list[Produto]:
    """Código/EAN exatos primeiro; depois descrição começando pelo termo; depois descrição
    contendo todas as palavras. Sem acento e sem diferenciar maiúsculas."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    consulta = select(Produto)
    if not incluir_inativos:
        consulta = consulta.where(Produto.ativo.is_(True))
    produtos = sorted(s.scalars(consulta), key=lambda p: (normalizar_texto(p.descricao), p.id))
    chave = normalizar_texto(termo)
    if not chave:
        return produtos
    palavras = chave.split()
    exatos: list[Produto] = []
    prefixos: list[Produto] = []
    contem: list[Produto] = []
    for p in produtos:
        if _casa_codigo(p, chave):
            exatos.append(p)
            continue
        descricao = normalizar_texto(p.descricao)
        if descricao.startswith(chave):
            prefixos.append(p)
        elif all(palavra in descricao for palavra in palavras):
            contem.append(p)
    return exatos + prefixos + contem


def listar_movimentos(s: Session, ator: Ator, produto_id: int) -> list[LinhaMovimento]:
    """Movimentos do produto, do mais recente para o mais antigo."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    _obter(s, produto_id)
    consulta = (
        select(
            MovimentoEstoque,
            Usuario.nome.label("usuario_nome"),
            NotaEntrada.numero.label("nota_numero"),
            NotaEntrada.serie.label("nota_serie"),
            NotaEntrada.emitente_nome.label("nota_emitente"),
        )
        .join(Usuario, MovimentoEstoque.usuario_id == Usuario.id)
        .outerjoin(NotaEntradaItem, MovimentoEstoque.nota_entrada_item_id == NotaEntradaItem.id)
        .outerjoin(NotaEntrada, NotaEntradaItem.nota_id == NotaEntrada.id)
        .where(MovimentoEstoque.produto_id == produto_id)
        .order_by(MovimentoEstoque.data_hora.desc(), MovimentoEstoque.id.desc())
    )
    linhas = []
    for row in s.execute(consulta):
        m = row.MovimentoEstoque
        nota = (
            None
            if row.nota_numero is None
            else f"NF {row.nota_numero}/{row.nota_serie} — {row.nota_emitente}"
        )
        linhas.append(
            LinhaMovimento(
                data_hora=m.data_hora,
                tipo=m.tipo,
                quantidade=m.quantidade,
                saldo_apos=m.saldo_apos,
                usuario_nome=row.usuario_nome,
                motivo=m.motivo,
                observacao=m.observacao,
                nota=nota,
            )
        )
    return linhas


def categorias_usadas(s: Session, ator: Ator) -> list[str]:
    exigir_modulo(ator, Modulo.ESTOQUE)
    valores = s.scalars(select(Produto.categoria).where(Produto.categoria != "").distinct())
    return sorted(valores, key=normalizar_texto)
```

```python
"""Bens Físicos: Ativo Não Circulante do Balancete. Só admin (módulo CFAUTO)."""

from decimal import Decimal
from typing import Any

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from cfauto.core import clock
from cfauto.core.audit import registrar
from cfauto.core.errors import RegistroNaoEncontrado, ValorInvalido
from cfauto.core.money import arredondar
from cfauto.core.permissoes import Ator, Modulo, exigir_modulo
from cfauto.core.sequencias import proximo_codigo
from cfauto.gestao.models import BemFisico
from cfauto.gestao.schemas import DadosBem


def _snapshot(bem: BemFisico) -> dict[str, Any]:
    return {
        "codigo": bem.codigo,
        "descricao": bem.descricao,
        "tipo": bem.tipo,
        "valor": bem.valor,
        "data_cadastro": bem.data_cadastro,
        "ativo": bem.ativo,
    }


def _validar(dados: DadosBem) -> tuple[str, Decimal]:
    descricao = dados.descricao.strip()
    if not descricao:
        raise ValorInvalido("Descrição obrigatória.")
    valor = arredondar(dados.valor)
    if valor <= 0:
        raise ValorInvalido("O valor deve ser maior que zero.")
    return descricao, valor


def _obter(s: Session, bem_id: int) -> BemFisico:
    bem = s.get(BemFisico, bem_id)
    if bem is None:
        raise RegistroNaoEncontrado("Bem não encontrado.")
    return bem


def criar_bem(s: Session, ator: Ator, dados: DadosBem) -> BemFisico:
    exigir_modulo(ator, Modulo.CFAUTO)
    descricao, valor = _validar(dados)
    bem = BemFisico(
        codigo=proximo_codigo(s, "bem", prefixo="BEM-", digitos=3),
        descricao=descricao,
        tipo=dados.tipo,
        valor=valor,
        data_cadastro=dados.data_cadastro or clock.hoje(),
        ativo=True,
    )
    s.add(bem)
    s.flush()
    registrar(s, ator.id, "criar", "bem_fisico", bem.id, None, _snapshot(bem))
    return bem


def atualizar_bem(s: Session, ator: Ator, bem_id: int, dados: DadosBem) -> BemFisico:
    exigir_modulo(ator, Modulo.CFAUTO)
    bem = _obter(s, bem_id)
    descricao, valor = _validar(dados)
    antes = _snapshot(bem)
    bem.descricao = descricao
    bem.tipo = dados.tipo
    bem.valor = valor
    if dados.data_cadastro is not None:
        bem.data_cadastro = dados.data_cadastro
    s.flush()
    registrar(s, ator.id, "alterar", "bem_fisico", bem.id, antes, _snapshot(bem))
    return bem


def _definir_ativo(s: Session, ator: Ator, bem_id: int, ativo: bool, acao: str) -> BemFisico:
    exigir_modulo(ator, Modulo.CFAUTO)
    bem = _obter(s, bem_id)
    if bem.ativo == ativo:
        raise ValorInvalido("O bem já está ativo." if ativo else "O bem já está baixado.")
    bem.ativo = ativo
    s.flush()
    registrar(s, ator.id, acao, "bem_fisico", bem.id, {"ativo": not ativo}, {"ativo": ativo})
    return bem


def baixar_bem(s: Session, ator: Ator, bem_id: int) -> BemFisico:
    return _definir_ativo(s, ator, bem_id, False, "baixar")


def reativar_bem(s: Session, ator: Ator, bem_id: int) -> BemFisico:
    return _definir_ativo(s, ator, bem_id, True, "reativar")


def listar_bens(s: Session, ator: Ator, *, incluir_baixados: bool = False) -> list[BemFisico]:
    exigir_modulo(ator, Modulo.CFAUTO)
    consulta = select(BemFisico).order_by(BemFisico.id)
    if not incluir_baixados:
        consulta = consulta.where(BemFisico.ativo.is_(True))
    return list(s.scalars(consulta))


def total_bens_ativos(s: Session, ator: Ator) -> Decimal:
    """Σ valor dos bens ativos (Ativo Não Circulante do Balancete)."""
    exigir_modulo(ator, Modulo.CFAUTO)
    total = s.scalar(select(func.sum(BemFisico.valor)).where(BemFisico.ativo.is_(True)))
    return total if total is not None else Decimal("0.00")
```

```python
"""Gera XML de NF-e 4.00 sintético para os testes (chave com DV válido)."""

from dataclasses import dataclass
from decimal import Decimal
from xml.sax.saxutils import escape

from cfauto.estoque.nfe import calcular_dv

CNPJ_FORNECEDOR = "12345678000195"
CNPJ_LOJA = "98765432000110"


@dataclass(frozen=True)
class ItemXml:
    codigo: str
    descricao: str
    qtd: str
    valor_produto: str
    ean: str = "SEM GTIN"
    ean_trib: str | None = None  # None = igual ao ean
    unidade: str = "UN"
    unidade_trib: str | None = None  # None = igual à unidade
    qtd_trib: str | None = None  # None = igual à qtd
    desconto: str | None = None
    frete: str | None = None
    ipi: str | None = None
    icms_st: str | None = None


def chave_teste(numero: int, cnpj: str = CNPJ_FORNECEDOR) -> str:
    base = f"312609{cnpj}55001{numero:09d}112345678"
    return base + str(calcular_dv(base))


def _tag(nome: str, valor: str | None) -> str:
    return "" if valor is None else f"<{nome}>{escape(valor)}</{nome}>"


def _det(numero: int, item: ItemXml) -> str:
    qtd = Decimal(item.qtd)
    valor_unit = Decimal(0) if qtd == 0 else Decimal(item.valor_produto) / qtd
    prod = (
        _tag("cProd", item.codigo)
        + _tag("cEAN", item.ean)
        + _tag("xProd", item.descricao)
        + _tag("uCom", item.unidade)
        + _tag("qCom", item.qtd)
        + _tag("vUnCom", f"{valor_unit:.10f}")
        + _tag("vProd", item.valor_produto)
        + _tag("cEANTrib", item.ean if item.ean_trib is None else item.ean_trib)
        + _tag("uTrib", item.unidade if item.unidade_trib is None else item.unidade_trib)
        + _tag("qTrib", item.qtd if item.qtd_trib is None else item.qtd_trib)
        + _tag("vFrete", item.frete)
        + _tag("vDesc", item.desconto)
        + "<indTot>1</indTot>"
    )
    if item.icms_st is None:
        icms = "<ICMSSN102><CSOSN>102</CSOSN></ICMSSN102>"
    else:
        icms = f"<ICMS10><CST>10</CST>{_tag('vICMSST', item.icms_st)}</ICMS10>"
    ipi = "" if item.ipi is None else f"<IPI><IPITrib>{_tag('vIPI', item.ipi)}</IPITrib></IPI>"
    imposto = f"<imposto><ICMS>{icms}</ICMS>{ipi}</imposto>"
    return f'<det nItem="{numero}"><prod>{prod}</prod>{imposto}</det>'


def _custo(item: ItemXml) -> Decimal:
    partes = (item.frete, item.ipi, item.icms_st)
    acrescimos = sum((Decimal(p) for p in partes if p is not None), Decimal(0))
    return Decimal(item.valor_produto) - Decimal(item.desconto or "0") + acrescimos


def montar_nfe(
    numero: int,
    itens: list[ItemXml],
    *,
    cnpj_emitente: str = CNPJ_FORNECEDOR,
    nome_emitente: str = "Distribuidora Boa Vista Ltda",
    cnpj_destinatario: str | None = CNPJ_LOJA,
    com_protocolo: bool = True,
) -> bytes:
    chave = chave_teste(numero, cnpj_emitente)
    total = sum((_custo(item) for item in itens), Decimal(0))
    destinatario = (
        ""
        if cnpj_destinatario is None
        else f"<dest>{_tag('CNPJ', cnpj_destinatario)}<xNome>Loja</xNome></dest>"
    )
    dets = "".join(_det(n, item) for n, item in enumerate(itens, start=1))
    nfe = (
        '<NFe xmlns="http://www.portalfiscal.inf.br/nfe">'
        f'<infNFe Id="NFe{chave}" versao="4.00">'
        f"<ide><cUF>31</cUF><mod>55</mod><serie>1</serie><nNF>{numero}</nNF>"
        "<dhEmi>2026-09-20T10:15:00-03:00</dhEmi></ide>"
        f"<emit>{_tag('CNPJ', cnpj_emitente)}{_tag('xNome', nome_emitente)}</emit>"
        f"{destinatario}{dets}"
        f"<total><ICMSTot><vNF>{total:.2f}</vNF></ICMSTot></total>"
        "</infNFe></NFe>"
    )
    if com_protocolo:
        corpo = (
            '<nfeProc xmlns="http://www.portalfiscal.inf.br/nfe" versao="4.00">'
            f"{nfe}<protNFe><infProt><chNFe>{chave}</chNFe><cStat>100</cStat></infProt>"
            "</protNFe></nfeProc>"
        )
    else:
        corpo = nfe
    return ('<?xml version="1.0" encoding="UTF-8"?>' + corpo).encode("utf-8")
```

```python
_TAG_NFE_PROC = "{http://www.portalfiscal.inf.br/nfe}nfeProc"
_TAG_NFE = "{http://www.portalfiscal.inf.br/nfe}NFe"

@dataclass(frozen=True)
class ItemNotaLido:
    n_item: int
    codigo: str
    ean: str | None
    ean_trib: str | None
    descricao: str
    unidade: str
    qtd: Decimal
    valor_unit: Decimal
    unidade_trib: str
    qtd_trib: Decimal
    valor_produto: Decimal
    desconto: Decimal
    frete: Decimal
    seguro: Decimal
    outros: Decimal
    ipi: Decimal
    icms_st: Decimal  # vICMSST + vFCPST

    @property
    def custo_total(self) -> Decimal:
        """Custo efetivo do item: o que a loja realmente pagou (decisão D2)."""
        return arredondar(
            self.valor_produto
            - self.desconto
            + self.frete
            + self.seguro
            + self.outros
            + self.ipi
            + self.icms_st

@dataclass(frozen=True)
class NotaFiscalLida:
    chave: str
    numero: int
    serie: int
    modelo: int
    data_emissao: date
    emitente_documento: str
    emitente_nome: str
    destinatario_documento: str | None
    valor_total: Decimal
    itens: tuple[ItemNotaLido, ...]


def calcular_dv(base: str) -> int:
    """Dígito verificador da chave (módulo 11, pesos 2..9 da direita para a esquerda)."""
    soma = sum(int(d) * _PESOS_DV[i % 8] for i, d in enumerate(reversed(base)))
    resto = soma % 11
    return 0 if resto < 2 else 11 - resto


def chave_valida(chave: str) -> bool:
    if len(chave) != 44 or not (chave.isascii() and chave.isdigit()):
        return False
    return calcular_dv(chave[:43]) == int(chave[43])


def ler_nfe(conteudo: bytes) -> NotaFiscalLida:
    try:
        raiz = XmlSeguro.fromstring(conteudo)
    except (XmlSeguro.ParseError, DefusedXmlException) as erro:
        raise XmlInvalido("O arquivo não é um XML válido.") from erro
    inf = _inf_nfe(raiz)
    chave = inf.get("Id", "").removeprefix("NFe")
    if not chave_valida(chave):
        raise XmlInvalido("Chave de acesso da NF-e inválida.")
    emissao = _texto(inf, "n:ide/n:dhEmi") or _obrigatorio(inf, "n:ide/n:dEmi", "dhEmi")
    try:
        data_emissao = para_data(emissao)
    except DataInvalida as erro:
        raise XmlInvalido(f"Data de emissão inválida: {emissao!r}.") from erro
    itens = tuple(_item(det) for det in inf.findall("n:det", _NS))
    if not itens:
        raise XmlInvalido("NF-e sem itens.")
    return NotaFiscalLida(
        chave=chave,
        numero=_inteiro(inf, "n:ide/n:nNF", "nNF"),
        serie=_inteiro(inf, "n:ide/n:serie", "serie"),
        modelo=_inteiro(inf, "n:ide/n:mod", "mod"),
        data_emissao=data_emissao,
        emitente_documento=_texto(inf, "n:emit/n:CNPJ")
        or _obrigatorio(inf, "n:emit/n:CPF", "CNPJ do emitente"),
        emitente_nome=_obrigatorio(inf, "n:emit/n:xNome", "nome do emitente"),
        destinatario_documento=_texto(inf, "n:dest/n:CNPJ") or _texto(inf, "n:dest/n:CPF"),
        valor_total=_decimal(inf, "n:total/n:ICMSTot/n:vNF", "vNF", obrigatorio=True),
        itens=itens,
    )


def _inf_nfe(raiz: Element) -> Element:
    if raiz.tag == _TAG_NFE_PROC:
        nfe = raiz.find("n:NFe", _NS)
    elif raiz.tag == _TAG_NFE:
        nfe = raiz
    else:
        nfe = None
    inf = None if nfe is None else nfe.find("n:infNFe", _NS)
    if inf is None:
        raise XmlInvalido("O arquivo não é o XML de uma NF-e (esperado nfeProc ou NFe).")
    return inf


def _item(det: Element) -> ItemNotaLido:
    n_item_texto = det.get("nItem", "")
    if not (n_item_texto.isascii() and n_item_texto.isdigit()):
        raise XmlInvalido("Item da NF-e sem número (nItem).")
    n_item = int(n_item_texto)
    prod = det.find("n:prod", _NS)
    if prod is None:
        raise XmlInvalido(f"Item {n_item} da NF-e sem dados do produto.")
    qtd = _decimal(prod, "n:qCom", "qCom", obrigatorio=True)
    if qtd <= 0:
        raise XmlInvalido(f"Item {n_item}: quantidade inválida ({qtd}).")
    unidade = _obrigatorio(prod, "n:uCom", "uCom")
    return ItemNotaLido(
        n_item=n_item,
        codigo=_obrigatorio(prod, "n:cProd", "cProd"),
        ean=_ean(_texto(prod, "n:cEAN")),
        ean_trib=_ean(_texto(prod, "n:cEANTrib")),
        descricao=_obrigatorio(prod, "n:xProd", "xProd"),
        unidade=unidade,
        qtd=qtd,
        valor_unit=_decimal(prod, "n:vUnCom", "vUnCom"),
        unidade_trib=_texto(prod, "n:uTrib") or unidade,
        qtd_trib=_decimal(prod, "n:qTrib", "qTrib") or qtd,
        valor_produto=_decimal(prod, "n:vProd", "vProd", obrigatorio=True),
        desconto=_decimal(prod, "n:vDesc", "vDesc"),
        frete=_decimal(prod, "n:vFrete", "vFrete"),
        seguro=_decimal(prod, "n:vSeg", "vSeg"),
        outros=_decimal(prod, "n:vOutro", "vOutro"),
        ipi=_decimal(det, "n:imposto/n:IPI/n:IPITrib/n:vIPI", "vIPI"),
        icms_st=_decimal(det, "n:imposto/n:ICMS/*/n:vICMSST", "vICMSST")
        + _decimal(det, "n:imposto/n:ICMS/*/n:vFCPST", "vFCPST"),
    )


def _texto(pai: Element, caminho: str) -> str | None:
    elemento = pai.find(caminho, _NS)
    if elemento is None or elemento.text is None:
        return None
    return elemento.text.strip() or None


def _obrigatorio(pai: Element, caminho: str, rotulo: str) -> str:
    valor = _texto(pai, caminho)
    if valor is None:
        raise XmlInvalido(f"NF-e sem o campo obrigatório {rotulo}.")
    return valor


def _decimal(pai: Element, caminho: str, rotulo: str, *, obrigatorio: bool = False) -> Decimal:
    texto = _obrigatorio(pai, caminho, rotulo) if obrigatorio else _texto(pai, caminho)
    if texto is None:
        return Decimal(0)
    try:
        valor = Decimal(texto)  # o XML usa ponto decimal: nunca para_decimal ("1.000" é 1)
    except InvalidOperation as erro:
        raise XmlInvalido(f"Valor inválido em {rotulo}: {texto!r}.") from erro
    if not valor.is_finite():
        raise XmlInvalido(f"Valor inválido em {rotulo}: {texto!r}.")
    return valor


def _inteiro(pai: Element, caminho: str, rotulo: str) -> int:
    texto = _obrigatorio(pai, caminho, rotulo)
    if not (texto.isascii() and texto.isdigit()):
        raise XmlInvalido(f"Valor inválido em {rotulo}: {texto!r}.")
    return int(texto)


def _ean(texto: str | None) -> str | None:
    """'SEM GTIN', vazio ou não numérico → None."""
    if texto is None or not (texto.isascii() and texto.isdigit()):
        return None
    return texto if len(texto) <= _TAMANHO_MAXIMO_EAN else None


# Port de extrairFator (01-ESTOQUE.gs:518-530), aplicado sobre o texto normalizado.
_RE_CONTAGENS = re.compile(
    r"(?:c/|cx\b|caixa\b)?\s*\d+\s*(?:capsulas|capsula|caps|cap|comprimidos|comprimido|comp"
    r"|drageas|dragea|softgel|gomas|goma|pastilhas|pastilha|saches|sache)\b"
)
_RE_MEDIDAS = re.compile(r"\b\d+\s*(?:kg|g|mg|mcg|l|ml)\b")
_RE_MULTIPLO = re.compile(r"(\d+)\s*x\s*\d+")
_RE_EMBALAGEM = re.compile(r"(?:c/|cx|caixa|fardo|pack|display)\s*(\d+)")
_RE_UNIDADES = re.compile(r"(\d+)\s*unidades?\b")


def fator_pela_descricao(descricao: str) -> int:
    """Unidades por embalagem lidas do nome ("C/12", "12X350ML", "PACK 6"); senão 1."""
    texto = _RE_MEDIDAS.sub("", _RE_CONTAGENS.sub("", normalizar_texto(descricao)))
    for regex in (_RE_MULTIPLO, _RE_EMBALAGEM, _RE_UNIDADES):
        achado = regex.search(texto)
        if achado is not None and int(achado.group(1)) > 0:
            return int(achado.group(1))
    return 1


def sugerir_fator(
    descricao: str, unidade: str, qtd: Decimal, unidade_trib: str, qtd_trib: Decimal
) -> Decimal:
    """Unidades de estoque por unidade comercial da nota. Só sugestão: o usuário confere."""
    if unidade_trib.strip().upper() != unidade.strip().upper() and qtd > 0:
        razao = qtd_trib / qtd
        if razao > 1 and razao == razao.to_integral_value():
            return razao.to_integral_value()
    return Decimal(fator_pela_descricao(descricao))
```

```python
"""Entrada de mercadoria por XML de NF-e: conferência e confirmação."""

from collections.abc import Sequence
from dataclasses import dataclass
from decimal import Decimal

from sqlalchemy import select
from sqlalchemy.orm import Session

from cfauto.core import clock, config
from cfauto.core.audit import registrar
from cfauto.core.errors import NotaJaImportada, ValorInvalido
from cfauto.core.money import arredondar, arredondar_custo, arredondar_qtd
from cfauto.core.permissoes import Ator, Modulo, exigir_modulo
from cfauto.estoque.models import (
    Destino,
    NotaEntrada,
    NotaEntradaItem,
    Produto,
    TipoMovimento,
    VinculoFornecedorProduto,
)
from cfauto.estoque.movimentos import movimentar
from cfauto.estoque.nfe import ItemNotaLido, NotaFiscalLida, ler_nfe, sugerir_fator
from cfauto.estoque.produtos import ean_em_uso, novo_codigo_produto, snapshot_produto
from cfauto.estoque.schemas import (
    AcaoItem,
    ConferenciaNota,
    DecisaoItem,
    ItemConferencia,
    OrigemCasamento,

def preparar_entrada(s: Session, ator: Ator, conteudo: bytes) -> ConferenciaNota:
    """Lê o XML e sugere o produto de cada item. Não grava nada."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    nota = ler_nfe(conteudo)
    _checar_duplicada(s, nota.chave)
    itens = tuple(_sugerir(s, nota.emitente_documento, item) for item in nota.itens)
    return ConferenciaNota(nota=nota, itens=itens, avisos=_avisos(s, nota))


def _avisos(s: Session, nota: NotaFiscalLida) -> tuple[str, ...]:
    configurado = str(config.obter(s, "empresa.cnpj", "") or "")
    cnpj_empresa = "".join(c for c in configurado if c.isdigit())
    destino = nota.destinatario_documento
    if cnpj_empresa and destino is not None and destino != cnpj_empresa:
        return (f"O destinatário da nota ({destino}) não é o CNPJ da empresa ({cnpj_empresa}).",)
    return ()


def _checar_duplicada(s: Session, chave: str) -> None:
    existente = s.scalar(
        select(NotaEntrada).where(NotaEntrada.chave == chave, NotaEntrada.estornada.is_(False))
    )
    if existente is not None:
        raise NotaJaImportada(
            f"A nota {existente.numero}/{existente.serie} de {existente.emitente_nome} "
            f"já foi lançada em {existente.data_entrada:%d/%m/%Y %H:%M}."
        )


def _vinculo(s: Session, cnpj: str, codigo: str) -> VinculoFornecedorProduto | None:
    return s.scalar(
        select(VinculoFornecedorProduto).where(
            VinculoFornecedorProduto.emitente_cnpj == cnpj,
            VinculoFornecedorProduto.codigo_fornecedor == codigo[:60],
        )
    )


def _produto_ativo_por_ean(s: Session, ean: str) -> Produto | None:
    produto = ean_em_uso(s, ean)
    return produto if produto is not None and produto.ativo else None


def _sugestao(
    item: ItemNotaLido,
    origem: OrigemCasamento,
    fator: Decimal,
    produto: Produto | None = None,
) -> ItemConferencia:
    return ItemConferencia(
        item=item,
        origem=origem,
        fator_sugerido=fator,
        produto_id=None if produto is None else produto.id,
        produto_codigo=None if produto is None else produto.codigo,
        produto_descricao=None if produto is None else produto.descricao,
    )


def _sugerir(s: Session, cnpj: str, item: ItemNotaLido) -> ItemConferencia:
    """Ordem: vínculo do fornecedor > EAN comercial > EAN tributável. Nunca por descrição."""
    vinculo = _vinculo(s, cnpj, item.codigo)
    if vinculo is not None:
        produto = s.get(Produto, vinculo.produto_id)
        if produto is not None and produto.ativo:
            return _sugestao(item, OrigemCasamento.VINCULO, vinculo.fator, produto)
    if item.ean is not None:
        produto = _produto_ativo_por_ean(s, item.ean)
        if produto is not None:
            # O EAN do produto identifica a embalagem dele: a unidade da nota é essa embalagem.
            return _sugestao(item, OrigemCasamento.EAN, produto.fator, produto)
    if item.ean_trib is not None and item.ean_trib != item.ean:
        produto = _produto_ativo_por_ean(s, item.ean_trib)
        if produto is not None:
            fator = arredondar_qtd(item.qtd_trib / item.qtd * produto.fator)
            return _sugestao(
                item, OrigemCasamento.EAN_TRIB, fator if fator > 0 else produto.fator, produto
            )
    fator = sugerir_fator(item.descricao, item.unidade, item.qtd, item.unidade_trib, item.qtd_trib)
    return _sugestao(item, OrigemCasamento.NENHUMA, fator)


@dataclass(frozen=True)
class _Plano:
    """Decisão de um item já validada contra o banco."""

    item: ItemNotaLido
    acao: AcaoItem
    fator: Decimal
    produto: Produto | None
    descricao_nova: str


def confirmar_entrada(
    s: Session, ator: Ator, conteudo: bytes, decisoes: Sequence[DecisaoItem]
) -> NotaEntrada:
    """Lança a nota: estoque, custo, data da última entrada, movimentos e vínculos."""
    exigir_modulo(ator, Modulo.ESTOQUE)
    lida = ler_nfe(conteudo)
    _checar_duplicada(s, lida.chave)
    planos = _validar_decisoes(s, lida, decisoes)  # tudo validado antes de gravar
    nota = NotaEntrada(
        chave=lida.chave,
        numero=lida.numero,
        serie=lida.serie,
        modelo=lida.modelo,
        emitente_cnpj=lida.emitente_documento,
        emitente_nome=lida.emitente_nome[:120],
        data_emissao=lida.data_emissao,
        data_entrada=clock.agora(),
        valor_total=arredondar(lida.valor_total),
        usuario_id=ator.id,
        xml_original=conteudo,
        estornada=False,
    )
    s.add(nota)
    s.flush()
    for plano in planos:
        registro = _registro_do_item(plano.item)
        nota.itens.append(registro)
        s.flush()
        if plano.acao is AcaoItem.IGNORAR:
            continue
        produto = plano.produto if plano.produto is not None else _criar_produto(s, ator, plano)
        _lancar(s, ator, lida.emitente_documento, registro, produto, plano)
    lancados = sum(1 for plano in planos if plano.acao is not AcaoItem.IGNORAR)
    registrar(
        s,
        ator.id,
        "entrada",
        "nota_entrada",
        nota.id,
        None,
        {
            "chave": nota.chave,
            "numero": nota.numero,
            "serie": nota.serie,
            "emitente": nota.emitente_nome,
            "emitente_cnpj": nota.emitente_cnpj,
            "valor_total": nota.valor_total,
            "itens_lancados": lancados,
            "itens_ignorados": len(planos) - lancados,
        },
    )
    return nota


def _validar_decisoes(
    s: Session, lida: NotaFiscalLida, decisoes: Sequence[DecisaoItem]
) -> list[_Plano]:
    por_item: dict[int, DecisaoItem] = {}
    for decisao in decisoes:
        if decisao.n_item in por_item:
            raise ValorInvalido(f"Item {decisao.n_item} decidido mais de uma vez.")
        por_item[decisao.n_item] = decisao
    numeros = {item.n_item for item in lida.itens}
    faltando = sorted(numeros - por_item.keys())
    if faltando:
        raise ValorInvalido("Itens sem decisão: " + ", ".join(map(str, faltando)) + ".")
    sobrando = sorted(por_item.keys() - numeros)
    if sobrando:
        raise ValorInvalido(
            "Itens que não existem na nota: " + ", ".join(map(str, sobrando)) + "."
        )
    return [_plano(s, item, por_item[item.n_item]) for item in lida.itens]


def _plano(s: Session, item: ItemNotaLido, decisao: DecisaoItem) -> _Plano:
    if decisao.acao is AcaoItem.IGNORAR:
        return _Plano(item, AcaoItem.IGNORAR, Decimal(0), None, "")
    fator = arredondar_qtd(decisao.fator) if decisao.fator is not None else Decimal(0)
    if fator <= 0:
        raise ValorInvalido(f"Item {item.n_item}: o fator deve ser maior que zero.")
    if arredondar_qtd(item.qtd * fator) <= 0:
        raise ValorInvalido(f"Item {item.n_item}: a quantidade no estoque ficaria zero.")
    if decisao.acao is AcaoItem.VINCULAR:
        produto = None if decisao.produto_id is None else s.get(Produto, decisao.produto_id)
        if produto is None or not produto.ativo:
            raise ValorInvalido(f"Item {item.n_item}: escolha um produto ativo.")
        return _Plano(item, AcaoItem.VINCULAR, fator, produto, "")
    descricao = (item.descricao if decisao.descricao is None else decisao.descricao).strip()
    if not descricao:
        raise ValorInvalido(f"Item {item.n_item}: informe a descrição do produto novo.")
    return _Plano(item, AcaoItem.CRIAR, fator, None, descricao)


def _registro_do_item(item: ItemNotaLido) -> NotaEntradaItem:
    return NotaEntradaItem(
        n_item=item.n_item,
        codigo_fornecedor=item.codigo[:60],
        ean=item.ean,
        descricao_nota=item.descricao[:200],
        unidade=item.unidade[:10],
        qtd_nota=item.qtd,
        valor_produto=arredondar(item.valor_produto),
        desconto=arredondar(item.desconto),
        frete=arredondar(item.frete),
        seguro=arredondar(item.seguro),
        outros=arredondar(item.outros),
        ipi=arredondar(item.ipi),
        icms_st=arredondar(item.icms_st),
        custo_total=item.custo_total,
    )


def _criar_produto(s: Session, ator: Ator, plano: _Plano) -> Produto:
    item = plano.item
    # EAN da nota só se ainda estiver livre (outro item da mesma nota pode tê-lo usado).
    ean = item.ean if item.ean is not None and ean_em_uso(s, item.ean) is None else None
    produto = Produto(
        codigo=novo_codigo_produto(s),
        ean=ean,
        descricao=plano.descricao_nova[:200],
        categoria="",
        embalagem=item.unidade[:30] or "Unidade",
        fator=plano.fator,
        preco_entrada=Decimal("0.0000"),
        preco_venda=Decimal("0.00"),
        estoque=Decimal("0.000"),
        estoque_minimo=Decimal("0.000"),
        estoque_minimo_manual=False,
        destino=Destino.REVENDA,
        markup_manual=False,
        ativo=True,
    )
    s.add(produto)
    s.flush()
    registrar(s, ator.id, "criar", "produto", produto.id, None, snapshot_produto(produto))
    return produto


def _lancar(
    s: Session,
    ator: Ator,
    cnpj: str,
    registro: NotaEntradaItem,
    produto: Produto,
    plano: _Plano,
) -> None:
    item = plano.item
    qtd_estoque = arredondar_qtd(item.qtd * plano.fator)
    registro.produto = produto
    registro.fator = plano.fator
    registro.qtd_estoque = qtd_estoque
    registro.preco_entrada_anterior = produto.preco_entrada
    registro.data_ult_entrada_anterior = produto.data_ult_entrada
    if item.custo_total > 0:
        # Custo por unidade de estoque, expresso na embalagem do produto (que a nota não muda).
        produto.preco_entrada = arredondar_custo(item.custo_total / qtd_estoque * produto.fator)
    # Bonificação (custo zero) soma estoque, mas não zera o custo do produto.
    registro.preco_entrada_aplicado = produto.preco_entrada
    produto.data_ult_entrada = clock.hoje()
    s.flush()
    movimentar(s, ator, produto, qtd_estoque, TipoMovimento.ENTRADA_NF, nota_item=registro)
    _gravar_vinculo(s, cnpj, item.codigo, produto, plano.fator)


def _gravar_vinculo(s: Session, cnpj: str, codigo: str, produto: Produto, fator: Decimal) -> None:
    vinculo = _vinculo(s, cnpj, codigo)
    if vinculo is None:
        s.add(
            VinculoFornecedorProduto(
                emitente_cnpj=cnpj,
                codigo_fornecedor=codigo[:60],
                produto_id=produto.id,
                fator=fator,
            )
        )
    else:
        vinculo.produto_id = produto.id
        vinculo.fator = fator
    s.flush()
```

```python
def _obter_nota(s: Session, nota_id: int) -> NotaEntrada:
    nota = s.get(NotaEntrada, nota_id)
    if nota is None:
        raise RegistroNaoEncontrado("Nota de entrada não encontrada.")
    return nota


@dataclass(frozen=True)
class _Lancado:
    item: NotaEntradaItem
    produto: Produto
    qtd: Decimal


def _lancados_em_ordem_inversa(nota: NotaEntrada) -> list[_Lancado]:
    """Itens com produto, do último para o primeiro: vários itens do mesmo produto
    restauram o preço na ordem certa."""
    resultado = []
    for item in sorted(nota.itens, key=lambda i: i.n_item, reverse=True):
        if item.produto_id is None:
            continue
        if item.produto is None or item.qtd_estoque is None:
            raise ValorInvalido(f"Item {item.n_item} da nota está inconsistente.")
        resultado.append(_Lancado(item, item.produto, item.qtd_estoque))
    return resultado


def _checar_saldos(lancados: list[_Lancado], confirmar_negativo: bool) -> None:
    a_estornar: dict[int, Decimal] = {}
    produtos: dict[int, Produto] = {}
    for lancado in lancados:
        pid = lancado.produto.id
        a_estornar[pid] = a_estornar.get(pid, Decimal(0)) + lancado.qtd
        produtos[pid] = lancado.produto
    negativos = [
        (produtos[pid], qtd) for pid, qtd in a_estornar.items() if produtos[pid].estoque - qtd < 0
    ]
    if negativos and not confirmar_negativo:
        detalhes = "; ".join(
            f"{p.codigo} — {p.descricao}: estoque {formatar_qtd(p.estoque)}, "
            f"estorno {formatar_qtd(qtd)}"
            for p, qtd in negativos
        )
        raise EstoqueInsuficiente(f"O estorno deixará o estoque negativo: {detalhes}.")


def _pode_restaurar(s: Session, nota: NotaEntrada, lancado: _Lancado) -> bool:
    """Só volta o preço/data se esta nota ainda é a última palavra sobre o custo do produto."""
    if lancado.produto.preco_entrada != lancado.item.preco_entrada_aplicado:
        return False  # entrada posterior ou edição manual mudou o custo
    posterior = s.scalar(
        select(NotaEntradaItem.id)
        .join(NotaEntrada, NotaEntradaItem.nota_id == NotaEntrada.id)
        .where(
            NotaEntradaItem.produto_id == lancado.produto.id,
            NotaEntrada.estornada.is_(False),
            NotaEntrada.id > nota.id,
        )
        .limit(1)
    )
    return posterior is None


def estornar_entrada(
    s: Session, ator: Ator, nota_id: int, *, confirmar_negativo: bool = False
) -> NotaEntrada:
    """Desfaz a nota (só admin). Nunca apaga produto; vínculos permanecem."""
    exigir_admin(ator)
    nota = _obter_nota(s, nota_id)
    if nota.estornada:
        raise ValorInvalido("Esta nota já foi estornada.")
    lancados = _lancados_em_ordem_inversa(nota)
    _checar_saldos(lancados, confirmar_negativo)
    for lancado in lancados:
        movimentar(
            s,
            ator,
            lancado.produto,
            -lancado.qtd,
            TipoMovimento.ESTORNO_NF,
            nota_item=lancado.item,
        )
        anterior = lancado.item.preco_entrada_anterior
        if anterior is not None and _pode_restaurar(s, nota, lancado):
            lancado.produto.preco_entrada = anterior
            lancado.produto.data_ult_entrada = lancado.item.data_ult_entrada_anterior
    nota.estornada = True
    nota.estornada_em = clock.agora()
    nota.estornada_por = ator.id
    s.flush()
    registrar(
        s,
        ator.id,
        "estornar",
        "nota_entrada",
        nota.id,
        {"estornada": False},
        {"estornada": True, "chave": nota.chave, "itens_estornados": len(lancados)},
    )
    return nota
```

```python
def criar_tabela(colunas: list[str]) -> QTableWidget:
    """Tabela só leitura, com seleção de uma linha inteira."""
    tabela = QTableWidget(0, len(colunas))
    tabela.setHorizontalHeaderLabels(colunas)
    tabela.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
    tabela.setSelectionMode(QAbstractItemView.SelectionMode.SingleSelection)
    tabela.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
    tabela.horizontalHeader().setStretchLastSection(True)

    tabela.clearSelection()
    tabela.setRowCount(len(linhas))

def ler_decimal(texto: str, rotulo: str) -> Decimal:
```

```python
def _celula(texto: str) -> QTableWidgetItem:
    celula = QTableWidgetItem(texto)
    celula.setFlags(celula.flags() & ~Qt.ItemFlag.ItemIsEditable)
    return celula


@dataclass
class _EstadoItem:
    item: ItemNotaLido
    acao: AcaoItem
    produto_id: int | None
    produto_texto: str
    descricao: str
    fator_texto: str


def _estado_inicial(sugestao: ItemConferencia) -> _EstadoItem:
    item = sugestao.item
    fator = formatar_numero(sugestao.fator_sugerido, 0)
    if sugestao.produto_id is None:
        return _EstadoItem(item, AcaoItem.CRIAR, None, "", item.descricao, fator)
    texto = f"{sugestao.produto_codigo} — {sugestao.produto_descricao}"
    return _EstadoItem(item, AcaoItem.VINCULAR, sugestao.produto_id, texto, item.descricao, fator)


class DialogoEscolherProduto(QDialog):
    """Busca de produto ativo para vincular a um item da nota."""

    def __init__(
        self, fabrica: sessionmaker[Session], ator: Ator, parent: QWidget | None = None
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._produtos: list[Produto] = []
        self.produto: Produto | None = None
        self.setWindowTitle("Escolher produto")
        self.resize(720, 420)
        self.busca = QLineEdit()
        self.busca.setPlaceholderText("Código, EAN ou descrição")
        self.busca.textChanged.connect(self._filtrar)
        self.tabela = criar_tabela(["Código", "EAN", "Descrição", "Embalagem", "Fator"])
        self.tabela.doubleClicked.connect(self._duplo_clique)
        botoes = QDialogButtonBox(
            QDialogButtonBox.StandardButton.Ok | QDialogButtonBox.StandardButton.Cancel
        )
        botoes.accepted.connect(self.escolher)
        botoes.rejected.connect(self.reject)
        layout = QVBoxLayout(self)
        layout.addWidget(self.busca)
        layout.addWidget(self.tabela)
        layout.addWidget(botoes)
        self._filtrar()

    def _filtrar(self) -> None:
        with transacao(self._fabrica) as s:
            self._produtos = buscar_produtos(s, self._ator, self.busca.text())
        preencher_tabela(
            self.tabela,
            [
                [p.codigo, p.ean or "", p.descricao, p.embalagem, formatar_qtd(p.fator)]
                for p in self._produtos
            ],
        )

    def _duplo_clique(self, _indice: QModelIndex) -> None:
        self.escolher()

    def escolher(self) -> None:
        linha = self.tabela.currentRow()
        if 0 <= linha < len(self._produtos):
            self.produto = self._produtos[linha]
            self.accept()


class DialogoConferencia(QDialog):
    """Conferência item a item antes de lançar a nota."""

    def __init__(
        self,
        fabrica: sessionmaker[Session],
        ator: Ator,
        conteudo: bytes,
        conferencia: ConferenciaNota,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._conteudo = conteudo
        self.nota_id: int | None = None
        nota = conferencia.nota
        self.setWindowTitle(f"Conferência da NF {nota.numero}/{nota.serie}")
        self.resize(1250, 520)
        cabecalho = QLabel(
            f"<b>{escape(nota.emitente_nome)}</b> — NF {nota.numero}/{nota.serie} — "
            f"emissão {nota.data_emissao:%d/%m/%Y} — total {formatar_brl(nota.valor_total)}"
        )
        self.avisos = QLabel("<br>".join(escape(aviso) for aviso in conferencia.avisos))
        self.avisos.setStyleSheet("color: #b9770e;")
        self.avisos.setVisible(bool(conferencia.avisos))
        self._estados = [_estado_inicial(sugestao) for sugestao in conferencia.itens]
        self._combos_acao: list[QComboBox] = []
        self._botoes_produto: list[QPushButton] = []
        self._campos_fator: list[QLineEdit] = []
        self.botao_confirmar = QPushButton("Confirmar entrada")
        self.botao_confirmar.clicked.connect(self.confirmar)
        self.tabela = QTableWidget(len(self._estados), len(_COLUNAS_CONFERENCIA))
        self.tabela.setHorizontalHeaderLabels(_COLUNAS_CONFERENCIA)
        self.tabela.verticalHeader().setVisible(False)
        self._montar_linhas()
        self.tabela.resizeColumnsToContents()
        self.tabela.itemChanged.connect(self._ao_editar_celula)

        cancelar = QPushButton("Cancelar")
        cancelar.clicked.connect(self.reject)
        rodape = QHBoxLayout()
        rodape.addStretch()
        rodape.addWidget(cancelar)
        rodape.addWidget(self.botao_confirmar)
        layout = QVBoxLayout(self)
        layout.addWidget(cabecalho)
        layout.addWidget(self.avisos)
        layout.addWidget(self.tabela)
        layout.addLayout(rodape)
        self._atualizar_confirmar()

    def _montar_linhas(self) -> None:
        for linha, estado in enumerate(self._estados):
            item = estado.item
            fixos = [
                str(item.n_item),
                item.codigo,
                item.ean or "",
                estado.descricao,
                item.unidade,
                formatar_qtd(item.qtd),
                formatar_brl(item.custo_total),
            ]
            for coluna, texto in enumerate(fixos):
                self.tabela.setItem(linha, coluna, _celula(texto))
            for coluna in (_COL_QTD_ESTOQUE, _COL_CUSTO_UNIT):
                self.tabela.setItem(linha, coluna, _celula(""))
            acao = QComboBox()
            for opcao in AcaoItem:
                acao.addItem(_TITULOS_ACAO[opcao], opcao)
            acao.setCurrentIndex(list(AcaoItem).index(estado.acao))
            acao.currentIndexChanged.connect(lambda _indice, n=linha: self._ao_mudar_acao(n))
            produto = QPushButton()
            produto.clicked.connect(lambda _marcado=False, n=linha: self._escolher_produto(n))
            fator = QLineEdit(estado.fator_texto)
            fator.textChanged.connect(lambda texto, n=linha: self._ao_mudar_fator(n, texto))
            self.tabela.setCellWidget(linha, _COL_ACAO, acao)
            self.tabela.setCellWidget(linha, _COL_PRODUTO, produto)
            self.tabela.setCellWidget(linha, _COL_FATOR, fator)
            self._combos_acao.append(acao)
            self._botoes_produto.append(produto)
            self._campos_fator.append(fator)
            self._atualizar_linha(linha)

    def _calculos(self, estado: _EstadoItem) -> tuple[str, str]:
        if estado.acao is AcaoItem.IGNORAR:
            return "—", "—"
        try:
            fator = ler_decimal(estado.fator_texto, "Fator")
        except ErroDominio:
            return "—", "—"  # fator incompleto durante a digitação
        qtd = arredondar_qtd(estado.item.qtd * fator)
        if qtd <= 0:
            return "—", "—"
        return formatar_qtd(qtd), formatar_brl(estado.item.custo_total / qtd)

    def _atualizar_linha(self, linha: int) -> None:
        estado = self._estados[linha]
        vincular = estado.acao is AcaoItem.VINCULAR
        botao = self._botoes_produto[linha]
        botao.setEnabled(vincular)
        if not vincular:
            botao.setText("—")
        elif estado.produto_id is None:
            botao.setText("Escolher…")
        else:
            botao.setText(estado.produto_texto)
        self._campos_fator[linha].setEnabled(estado.acao is not AcaoItem.IGNORAR)
        descricao = self.tabela.item(linha, _COL_DESCRICAO)
        if estado.acao is AcaoItem.CRIAR:
            descricao.setFlags(descricao.flags() | Qt.ItemFlag.ItemIsEditable)
        else:
            descricao.setFlags(descricao.flags() & ~Qt.ItemFlag.ItemIsEditable)
        qtd, custo = self._calculos(estado)
        self.tabela.item(linha, _COL_QTD_ESTOQUE).setText(qtd)
        self.tabela.item(linha, _COL_CUSTO_UNIT).setText(custo)

    def _ao_mudar_acao(self, linha: int) -> None:
        self._estados[linha].acao = AcaoItem(self._combos_acao[linha].currentData())
        self._atualizar_linha(linha)
        self._atualizar_confirmar()

    def _ao_mudar_fator(self, linha: int, texto: str) -> None:
        self._estados[linha].fator_texto = texto
        self._atualizar_linha(linha)

    def _ao_editar_celula(self, celula: QTableWidgetItem) -> None:
        if celula.column() == _COL_DESCRICAO:
            self._estados[celula.row()].descricao = celula.text()

    def _escolher_produto(self, linha: int) -> None:
        dialogo = DialogoEscolherProduto(self._fabrica, self._ator, self)
        if dialogo.exec() and dialogo.produto is not None:
            self.definir_produto(linha, dialogo.produto)

    def acao_da_linha(self, linha: int) -> AcaoItem:
        return self._estados[linha].acao

    def campo_fator(self, linha: int) -> QLineEdit:
        return self._campos_fator[linha]

    def definir_acao(self, linha: int, acao: AcaoItem) -> None:
        self._combos_acao[linha].setCurrentIndex(list(AcaoItem).index(acao))

    def definir_produto(self, linha: int, produto: Produto) -> None:
        """Vincula o item ao produto escolhido; o fator passa a ser o do produto."""
        estado = self._estados[linha]
        estado.produto_id = produto.id
        estado.produto_texto = f"{produto.codigo} — {produto.descricao}"
        self.definir_acao(linha, AcaoItem.VINCULAR)
        self._campos_fator[linha].setText(formatar_numero(produto.fator, 0))
        self._atualizar_linha(linha)
        self._atualizar_confirmar()

    def pronto(self) -> bool:
        return all(
            estado.acao is not AcaoItem.VINCULAR or estado.produto_id is not None
            for estado in self._estados
        )

    def _atualizar_confirmar(self) -> None:
        self.botao_confirmar.setEnabled(self.pronto())

    def decisoes(self) -> list[DecisaoItem]:
        resultado = []
        for estado in self._estados:
            numero = estado.item.n_item
            if estado.acao is AcaoItem.IGNORAR:
                resultado.append(DecisaoItem(numero, AcaoItem.IGNORAR))
                continue
            fator = ler_decimal(estado.fator_texto, f"Fator do item {numero}")
            if estado.acao is AcaoItem.VINCULAR:
                resultado.append(
                    DecisaoItem(
                        numero, AcaoItem.VINCULAR, produto_id=estado.produto_id, fator=fator
                    )
                )
            else:
                resultado.append(
                    DecisaoItem(numero, AcaoItem.CRIAR, descricao=estado.descricao, fator=fator)
                )
        return resultado

    def confirmar(self) -> None:
        try:
            decisoes = self.decisoes()
            with transacao(self._fabrica) as s:
                self.nota_id = confirmar_entrada(s, self._ator, self._conteudo, decisoes).id
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.accept()


def _linha_item(item: NotaEntradaItem) -> list[str]:
    produto = (
        "Ignorado" if item.produto is None else f"{item.produto.codigo} — {item.produto.descricao}"
    )
    return [
        str(item.n_item),
        item.descricao_nota,
        formatar_qtd(item.qtd_nota),
        formatar_brl(item.custo_total),
        produto,
        "" if item.fator is None else formatar_qtd(item.fator),
        "" if item.qtd_estoque is None else formatar_qtd(item.qtd_estoque),
    ]

class TelaEntradas(QWidget):
    def __init__(
        self, fabrica: sessionmaker[Session], ator: Ator, parent: QWidget | None = None
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._notas: list[NotaEntrada] = []

        importar = QPushButton("Importar XML…")
        importar.clicked.connect(self._escolher_arquivo)
        itens = QPushButton("Ver itens")
        itens.clicked.connect(self._ver_itens)
        self.botao_estornar = QPushButton("Estornar")
        self.botao_estornar.clicked.connect(self._estornar_selecionada)
        self.botao_estornar.setVisible(ator.admin)
        self.tabela = criar_tabela(_COLUNAS_NOTAS)
        self.tabela.doubleClicked.connect(self._duplo_clique)

        botoes = QHBoxLayout()
        for botao in (importar, itens, self.botao_estornar):
            botoes.addWidget(botao)
        botoes.addStretch()
        layout = QVBoxLayout(self)
        layout.addLayout(botoes)
        layout.addWidget(self.tabela)
        self.recarregar()

    def recarregar(self) -> None:
        with transacao(self._fabrica) as s:
            self._notas = listar_notas(s, self._ator)
        preencher_tabela(self.tabela, [_linha_nota(nota) for nota in self._notas])

    def ids_na_tabela(self) -> list[int]:
        return [nota.id for nota in self._notas]

    def selecionar(self, linha: int) -> None:
        self.tabela.setCurrentCell(linha, 0)

    def selecionada(self) -> NotaEntrada | None:
        linha = self.tabela.currentRow()
        return self._notas[linha] if 0 <= linha < len(self._notas) else None

    def _escolher_arquivo(self) -> None:
        caminho, _filtro = QFileDialog.getOpenFileName(
            self, "Importar XML da NF-e", "", "XML da NF-e (*.xml)"
        )
        if caminho:
            self.importar_arquivo(Path(caminho))

    def importar_arquivo(self, caminho: Path) -> None:
        try:
            conteudo = caminho.read_bytes()
        except OSError as erro:
            mostrar_erro(self, ValorInvalido(f"Não foi possível ler o arquivo: {erro}"))
            return
        try:
            with transacao(self._fabrica) as s:
                conferencia = preparar_entrada(s, self._ator, conteudo)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        if DialogoConferencia(self._fabrica, self._ator, conteudo, conferencia, self).exec():
            self.recarregar()

    def _ver_itens(self) -> None:
        nota = self.selecionada()
        if nota is not None:
            DialogoItensNota(self._fabrica, self._ator, nota.id, self).exec()

    def _duplo_clique(self, _indice: QModelIndex) -> None:
        self._ver_itens()

    def _estornar_selecionada(self) -> None:
        nota = self.selecionada()
        if nota is not None:
            self.estornar_nota(nota.id, self._perguntar)

    def _perguntar(self, texto: str) -> bool:
        resposta = QMessageBox.question(self, "CFAUTO", texto)
        return resposta == QMessageBox.StandardButton.Yes

    def _estornar(self, nota_id: int, *, negativo: bool) -> None:
        with transacao(self._fabrica) as s:
            estornar_entrada(s, self._ator, nota_id, confirmar_negativo=negativo)

    def estornar_nota(self, nota_id: int, confirmar: Callable[[str], bool]) -> bool:
        """Estorna com confirmação; se o saldo ficar negativo, pede uma segunda confirmação."""
        if not confirmar("Estornar esta nota? As quantidades lançadas saem do estoque."):
            return False
        try:
            try:
                self._estornar(nota_id, negativo=False)
            except EstoqueInsuficiente as erro:
                if not confirmar(f"{erro}\n\nEstornar mesmo assim?"):
                    return False
                self._estornar(nota_id, negativo=True)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return False
        self.recarregar()
        return True
```

```python
        botoes = QDialogButtonBox(
            QDialogButtonBox.StandardButton.Save | QDialogButtonBox.StandardButton.Cancel
        )
        botoes.accepted.connect(self.salvar)
        botoes.rejected.connect(self.reject)
        layout = QVBoxLayout(self)
        layout.addLayout(form)
        layout.addWidget(botoes)

    def salvar(self) -> None:
        try:
            dados = DadosBem(
                descricao=self.descricao.text(),
                tipo=TipoBem(self.tipo.currentData()),
                valor=ler_decimal(self.valor.text(), "Valor"),

            )

                if self._bem is None:
                    criar_bem(s, self._ator, dados)
                else:
                    atualizar_bem(s, self._ator, self._bem.id, dados)
        except ErroDominio as erro:

class TelaBens(QWidget):
    def __init__(
        self, fabrica: sessionmaker[Session], ator: Ator, parent: QWidget | None = None
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._bens: list[BemFisico] = []

        self.mostrar_baixados = QCheckBox("Mostrar baixados")
        self.mostrar_baixados.toggled.connect(self.recarregar)
        novo = QPushButton("Novo")
        novo.clicked.connect(self._novo)
        editar = QPushButton("Editar")
        editar.clicked.connect(self._editar)
        self.botao_situacao = QPushButton("Baixar")
        self.botao_situacao.clicked.connect(self.alternar_situacao)
        self.tabela = criar_tabela(_COLUNAS)
        self.tabela.doubleClicked.connect(self._editar_pelo_duplo_clique)
        self.tabela.itemSelectionChanged.connect(self._atualizar_botao_situacao)
        self.total_rotulo = QLabel()

        botoes = QHBoxLayout()
        for botao in (novo, editar, self.botao_situacao):
            botoes.addWidget(botao)
        botoes.addStretch()
        botoes.addWidget(self.mostrar_baixados)
        layout = QVBoxLayout(self)
        layout.addLayout(botoes)
        layout.addWidget(self.tabela)
        layout.addWidget(self.total_rotulo)
        self.recarregar()

    def recarregar(self) -> None:
        with transacao(self._fabrica) as s:
            self._bens = listar_bens(
                s, self._ator, incluir_baixados=self.mostrar_baixados.isChecked()
            )
            total = total_bens_ativos(s, self._ator)
        preencher_tabela(
            self.tabela,
            [
                [
                    bem.codigo,
                    bem.descricao,
                    TITULOS_TIPO_BEM[bem.tipo],
                    formatar_brl(bem.valor),
                    f"{bem.data_cadastro:%d/%m/%Y}",
                    "Ativo" if bem.ativo else "Baixado",
                ]
                for bem in self._bens
            ],
        )
        self.total_rotulo.setText(f"Total dos bens ativos: {formatar_brl(total)}")
        self._atualizar_botao_situacao()

    def codigos_na_tabela(self) -> list[str]:
        return [bem.codigo for bem in self._bens]

    def selecionar(self, linha: int) -> None:
        self.tabela.setCurrentCell(linha, 0)

    def selecionado(self) -> BemFisico | None:
        linha = self.tabela.currentRow()
        return self._bens[linha] if 0 <= linha < len(self._bens) else None

    def _novo(self) -> None:
        if DialogoBem(self._fabrica, self._ator, parent=self).exec():
            self.recarregar()

    def _editar(self) -> None:
        bem = self.selecionado()
        if bem is not None and DialogoBem(self._fabrica, self._ator, bem, self).exec():
            self.recarregar()

    def _editar_pelo_duplo_clique(self, _indice: QModelIndex) -> None:
        self._editar()

    def alternar_situacao(self) -> None:
        bem = self.selecionado()
        if bem is None:
            return
        try:
            with transacao(self._fabrica) as s:
                if bem.ativo:
                    baixar_bem(s, self._ator, bem.id)
                else:
                    reativar_bem(s, self._ator, bem.id)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.recarregar()

    def _atualizar_botao_situacao(self) -> None:
        bem = self.selecionado()
        baixado = bem is not None and not bem.ativo
        self.botao_situacao.setText("Reativar" if baixado else "Baixar")
```
