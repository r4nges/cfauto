## Global Constraints

- Idioma do código de domínio, mensagens e UI: português (identificadores sem acento: `cliente`, `lancar`). Mensagens ao usuário com acentuação correta.
- Python 3.13 via uv; rode tudo na raiz `C:\Users\Admin\Documents\CFAUTO` com `uv run ...`. O hook RTK esconde o resumo do pytest: use `rtk proxy uv run pytest ...` para ver a saída real.
- Dinheiro e percentuais: `Decimal`, nunca `float`. Colunas `Dinheiro()` (2 casas) e `DecimalEscalado(2)`; enums via `EnumTexto(Enum)`.
- Data atual só por `cfauto.core.clock.agora()` / `clock.hoje()`. Datas digitadas passam por `cfauto.core.dates.para_data`.
- Acesso a dados sempre por nome de atributo; consultas com colunas soltas usam `.label()`.
- `ui/` nunca monta consulta/SQL; chama serviços de `cfauto/clientes/` (e `cfauto.core.config.obter`). Serviços de Clientes devolvem dataclasses de `cfauto/clientes/schemas.py`, nunca entidade ORM (exceção: `carregar_cliente` e `razao.lancar`, internos ao domínio).
- UI: valores digitados em `QLineEdit` convertidos com `cfauto.ui.comum.ler_decimal`; nunca `QDoubleSpinBox`. Combos guardam o enum como dado e a leitura reconverte (`Sexo(combo.currentData())`).
- Permissões: tudo `exigir_modulo(ator, Modulo.CLIENTES)`; limite diferente do padrão na criação ou alteração de limite exige admin; inativar/reativar/excluir `exigir_admin(ator)`.
- `cliente.saldo_devedor` é alterado **somente** por `cfauto.clientes.razao.lancar`. `cliente_lancamento` nunca é atualizado nem apagado.
- Nenhuma exceção é engolida devolvendo valor padrão; erros de regra são subclasses de `cfauto.core.errors.ErroDominio`. Auditoria via `cfauto.core.audit.registrar(s, ator.id, acao, "cliente", id, antes, depois)` na mesma transação (ação e entidade separadas, como no resto do código).
- Testes de UI que exercitam erro fazem `monkeypatch` de `mostrar_erro` no módulo da tela; confirmações recebem `confirmar=` (o `QMessageBox` bloqueia o teste offscreen).
- Antes do `uv run ruff check .`, rode `uv run ruff format <arquivos da task>`. mypy: `uv run -m mypy` (o Smart App Control bloqueia `mypy.exe`).
- Commits: Conventional Commits, terminando com as linhas de attribution do SEU system-reminder.

```python
"""Clientes: cadastro, dependentes e razão (livro) de lançamentos do saldo devedor."""

from datetime import date, datetime
from decimal import Decimal
from enum import StrEnum

from sqlalchemy import ForeignKey, Index, String
from sqlalchemy.orm import Mapped, mapped_column, relationship

from cfauto.core.db import Base, ComTimestamps
from cfauto.core.tipos import DecimalEscalado, Dinheiro, EnumTexto

LIMITE_PADRAO = Decimal("500.00")
MAX_DEPENDENTES = 4


class Sexo(StrEnum):
    M = "M"
    F = "F"
    OUTRO = "OUTRO"


TITULOS_SEXO: dict[Sexo, str] = {Sexo.M: "Masculino", Sexo.F: "Feminino", Sexo.OUTRO: "Outro"}


class FrequenciaJuros(StrEnum):
    DIARIO = "DIARIO"
    MENSAL = "MENSAL"
    BIMESTRAL = "BIMESTRAL"


TITULOS_FREQUENCIA: dict[FrequenciaJuros, str] = {
    FrequenciaJuros.DIARIO: "Diária",
    FrequenciaJuros.MENSAL: "Mensal",
    FrequenciaJuros.BIMESTRAL: "Bimestral",
}


class TipoLancamento(StrEnum):
    DEBITO = "DEBITO"  # aumenta a dívida
    CREDITO = "CREDITO"  # abate a dívida


class MotivoLancamento(StrEnum):
    COMPRA_ANOTADA = "COMPRA_ANOTADA"
    PAGAMENTO = "PAGAMENTO"
    CORRECAO = "CORRECAO"
    OUTRO = "OUTRO"


TITULOS_MOTIVO_LANCAMENTO: dict[MotivoLancamento, str] = {
    MotivoLancamento.COMPRA_ANOTADA: "Compra anotada",
    MotivoLancamento.PAGAMENTO: "Pagamento",
    MotivoLancamento.CORRECAO: "Correção",
    MotivoLancamento.OUTRO: "Outro",
}


class Cliente(ComTimestamps, Base):
    __tablename__ = "cliente"

    id: Mapped[int] = mapped_column(primary_key=True)
    codigo: Mapped[str] = mapped_column(String(20), unique=True)
    nome: Mapped[str] = mapped_column(String(120))
    nome_busca: Mapped[str] = mapped_column(String(120), index=True)
    celular: Mapped[str] = mapped_column(String(11), index=True)
    cpf: Mapped[str | None] = mapped_column(String(11), index=True)
    email: Mapped[str | None] = mapped_column(String(120))
    endereco: Mapped[str] = mapped_column(String(200), default="")
    nascimento: Mapped[date | None]
    sexo: Mapped[Sexo | None] = mapped_column(EnumTexto(Sexo))
    observacoes: Mapped[str] = mapped_column(String(500), default="")
    # Nulo = sem limite.
    limite: Mapped[Decimal | None] = mapped_column(Dinheiro())
    # Só muda por razao.lancar(); negativo = crédito a favor do cliente.
    saldo_devedor: Mapped[Decimal] = mapped_column(Dinheiro(), default=Decimal(0))
    somente_titular: Mapped[bool] = mapped_column(default=False)
    # Guardados sem efeito, como no original.
    aumento_auto: Mapped[bool] = mapped_column(default=False)
    aumento_auto_pct: Mapped[Decimal] = mapped_column(DecimalEscalado(2), default=Decimal(5))
    juros_dias: Mapped[int] = mapped_column(default=30)
    juros_frequencia: Mapped[FrequenciaJuros] = mapped_column(
        EnumTexto(FrequenciaJuros), default=FrequenciaJuros.MENSAL
    )
    juros_pct: Mapped[Decimal] = mapped_column(DecimalEscalado(2), default=Decimal(2))
    juros_mora: Mapped[bool] = mapped_column(default=False)
    ativo: Mapped[bool] = mapped_column(default=True)
    dependentes: Mapped[list["ClienteDependente"]] = relationship(
        back_populates="cliente",
        cascade="all, delete-orphan",
        order_by="ClienteDependente.id",
        lazy="selectin",
    )

    @property
    def disponivel(self) -> Decimal | None:
        """None = ilimitado. Pode passar do limite (crédito a favor) ou ficar negativo."""
        return None if self.limite is None else self.limite - self.saldo_devedor


class ClienteDependente(ComTimestamps, Base):
    __tablename__ = "cliente_dependente"

    id: Mapped[int] = mapped_column(primary_key=True)
    cliente_id: Mapped[int] = mapped_column(ForeignKey("cliente.id"), index=True)
    nome: Mapped[str] = mapped_column(String(120))
    parentesco: Mapped[str] = mapped_column(String(40), default="")
    cliente: Mapped[Cliente] = relationship(back_populates="dependentes")


class ClienteLancamento(ComTimestamps, Base):
    """Livro do saldo devedor: só inserção, nunca alterado nem apagado."""

    __tablename__ = "cliente_lancamento"
    __table_args__ = (Index("ix_cliente_lancamento_cliente_data", "cliente_id", "data_hora"),)

    id: Mapped[int] = mapped_column(primary_key=True)
    cliente_id: Mapped[int] = mapped_column(ForeignKey("cliente.id"))
    data_hora: Mapped[datetime]
    tipo: Mapped[TipoLancamento] = mapped_column(EnumTexto(TipoLancamento))
    valor: Mapped[Decimal] = mapped_column(Dinheiro())
    saldo_apos: Mapped[Decimal] = mapped_column(Dinheiro())
    motivo: Mapped[MotivoLancamento] = mapped_column(EnumTexto(MotivoLancamento))
    observacao: Mapped[str] = mapped_column(String(200), default="")
    usuario_id: Mapped[int] = mapped_column(ForeignKey("usuario.id"))
```

```python
"""Validação e formatação dos dados de cliente (funções puras, sem banco)."""

import re
from decimal import Decimal

from cfauto.core.errors import ValorInvalido
from cfauto.core.money import formatar_brl

TAMANHO_NOME = 120
TAMANHO_EMAIL = 120
_EMAIL = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")


def so_digitos(texto: str) -> str:
    return "".join(c for c in texto if c.isascii() and c.isdigit())


def normalizar_nome(texto: str) -> str:
    nome = " ".join(texto.split())
    if len(nome.split()) < 2:
        raise ValorInvalido("Informe nome e sobrenome.")
    if len(nome) > TAMANHO_NOME:
        raise ValorInvalido(f"Nome: máximo de {TAMANHO_NOME} caracteres.")
    return nome


def normalizar_celular(texto: str) -> str:
    """Celular com DDD, só dígitos (10 ou 11). Aceita o prefixo +55."""
    digitos = so_digitos(texto)
    if not digitos:
        raise ValorInvalido("Celular obrigatório.")
    if len(digitos) in (12, 13) and digitos.startswith("55"):
        digitos = digitos[2:]
    if len(digitos) not in (10, 11):
        raise ValorInvalido("Celular deve ter DDD + número (10 ou 11 dígitos).")
    if int(digitos[:2]) < 11:
        raise ValorInvalido("DDD inválido.")
    return digitos


def cpf_valido(digitos: str) -> bool:
    if len(digitos) != 11 or not (digitos.isascii() and digitos.isdigit()):
        return False
    if len(set(digitos)) == 1:
        return False
    for tamanho in (9, 10):
        pesos = range(tamanho + 1, 1, -1)
        soma = sum(int(d) * peso for d, peso in zip(digitos[:tamanho], pesos, strict=True))
        if (soma * 10) % 11 % 10 != int(digitos[tamanho]):
            return False
    return True


def normalizar_cpf(texto: str) -> str | None:
    """Vazio (ou só pontuação de máscara) → None; senão 11 dígitos com DV válido."""
    if not texto.strip(" .-"):
        return None
    digitos = so_digitos(texto)
    if not cpf_valido(digitos):
        raise ValorInvalido("CPF inválido.")
    return digitos


def normalizar_email(texto: str) -> str | None:
    email = texto.strip()
    if not email:
        return None
    if len(email) > TAMANHO_EMAIL or not _EMAIL.match(email):
        raise ValorInvalido("E-mail inválido.")
    return email


def formatar_celular(digitos: str) -> str:
    if len(digitos) == 11:
        return f"({digitos[:2]}) {digitos[2:7]}-{digitos[7:]}"
    if len(digitos) == 10:
        return f"({digitos[:2]}) {digitos[2:6]}-{digitos[6:]}"
    return digitos


def formatar_cpf(digitos: str | None) -> str:
    if not digitos:
        return ""
    return f"{digitos[:3]}.{digitos[3:6]}.{digitos[6:9]}-{digitos[9:]}"


def formatar_saldo(saldo: Decimal) -> str:
    """Saldo devedor para exibição; negativo é crédito a favor do cliente."""
    if saldo < 0:
        return f"Crédito a favor {formatar_brl(-saldo)}"
    return formatar_brl(saldo)
```

```python
    if dados.nascimento is not None and dados.nascimento > clock.hoje():

        raise ValorInvalido("Os parâmetros de juros não podem ser negativos.")
    if len(dados.dependentes) > MAX_DEPENDENTES:
        raise ValorInvalido(f"No máximo {MAX_DEPENDENTES} dependentes.")
    if dados.somente_titular and dados.dependentes:
        raise ValorInvalido("Cliente marcado como somente titular não pode ter dependentes.")
    dependentes: list[tuple[str, str]] = []
    for dependente in dados.dependentes:
        nome_dependente = " ".join(_texto(dependente.nome, "Nome do dependente", 120).split())
        if not nome_dependente:
            raise ValorInvalido("Nome do dependente obrigatório.")
        dependentes.append((nome_dependente, _texto(dependente.parentesco, "Parentesco", 40)))
    return valores, dependentes

def criar_cliente(s: Session, ator: Ator, dados: DadosCliente) -> int:
    exigir_modulo(ator, Modulo.CLIENTES)
    valores, dependentes = _validar(dados)
    if valores["limite"] != LIMITE_PADRAO:
        _exigir_admin_para_limite(ator)
    cliente = Cliente(
        codigo=proximo_codigo(s, "cliente", prefixo="CLI-", digitos=4),
        saldo_devedor=Decimal("0.00"),
        ativo=True,
        **valores,
    )
    cliente.dependentes = _novos_dependentes(dependentes)
    s.add(cliente)
    s.flush()
    registrar(s, ator.id, "criar", "cliente", cliente.id, None, _snapshot(cliente))
    return cliente.id


def editar_cliente(s: Session, ator: Ator, cliente_id: int, dados: DadosCliente) -> None:
    """Nunca altera código nem saldo devedor; a lista de dependentes é substituída inteira."""

    cliente = carregar_cliente(s, cliente_id)
    valores, dependentes = _validar(dados)
    if valores["limite"] != cliente.limite:
        _exigir_admin_para_limite(ator)
    antes = _snapshot(cliente)
    for campo, valor in valores.items():
        setattr(cliente, campo, valor)
    cliente.dependentes = _novos_dependentes(dependentes)
    s.flush()
    registrar(s, ator.id, "alterar", "cliente", cliente.id, antes, _snapshot(cliente))


def verificar_duplicados(
    s: Session, ator: Ator, dados: DadosCliente, ignorar_id: int | None = None

    aparecer primeiro."""

    if ignorar_id is not None:

    return [resumo_cliente(c) for c in s.scalars(consulta)]


def buscar_clientes(
    s: Session, ator: Ator, texto: str = "", *, incluir_inativos: bool = False

    """Por nome (sem acento), código, ou dígitos de CPF/celular (a partir de 3 dígitos)."""

    return [resumo_cliente(c) for c in s.scalars(consulta)]
```

```python
"""Razão do cliente: única porta de alteração do saldo devedor, lançamento manual e extrato."""

from datetime import date, datetime, time, timedelta
from decimal import Decimal

from sqlalchemy import select
from sqlalchemy.orm import Session

from cfauto.auth.models import Usuario
from cfauto.clientes.cadastro import carregar_cliente, resumo_cliente
from cfauto.clientes.models import Cliente, ClienteLancamento, MotivoLancamento, TipoLancamento
from cfauto.clientes.schemas import Extrato, LinhaLancamento
from cfauto.core import clock
from cfauto.core.audit import registrar
from cfauto.core.errors import ValorInvalido
from cfauto.core.money import arredondar
from cfauto.core.permissoes import Ator, Modulo, exigir_modulo

TAMANHO_OBSERVACAO = 200


def saldo_apos(saldo: Decimal, tipo: TipoLancamento, valor: Decimal) -> Decimal:
    """DEBITO aumenta a dívida; CREDITO abate (pode ficar negativo: crédito a favor)."""
    return saldo + valor if tipo == TipoLancamento.DEBITO else saldo - valor


def lancar(
    s: Session,
    ator: Ator,
    cliente: Cliente,
    tipo: TipoLancamento,
    valor: Decimal,
    motivo: MotivoLancamento,
    observacao: str = "",
) -> ClienteLancamento:
    """Única porta de alteração de `saldo_devedor`. Não checa permissão, limite nem situação
    do cliente: quem chama decide."""
    valor = arredondar(valor)
    if valor <= 0:
        raise ValorInvalido("O valor do lançamento deve ser maior que zero.")
    observacao = observacao.strip()
    if len(observacao) > TAMANHO_OBSERVACAO:
        raise ValorInvalido(f"Observação: máximo de {TAMANHO_OBSERVACAO} caracteres.")
    if motivo == MotivoLancamento.CORRECAO and not observacao:
        raise ValorInvalido("Informe na observação o motivo da correção.")
    antes = cliente.saldo_devedor
    cliente.saldo_devedor = saldo_apos(antes, tipo, valor)
    lancamento = ClienteLancamento(
        cliente_id=cliente.id,
        data_hora=clock.agora(),
        tipo=tipo,
        valor=valor,
        saldo_apos=cliente.saldo_devedor,
        motivo=motivo,
        observacao=observacao,
        usuario_id=ator.id,
    )
    s.add(lancamento)
    s.flush()
    registrar(
        s,
        ator.id,
        "lancamento",
        "cliente",
        cliente.id,
        {"saldo_devedor": antes},
        {
            "saldo_devedor": cliente.saldo_devedor,
            "tipo": tipo,
            "valor": valor,
            "motivo": motivo,
            "lancamento_id": lancamento.id,
        },
    )
    return lancamento


def _linha(lancamento: ClienteLancamento, usuario: str) -> LinhaLancamento:
    return LinhaLancamento(
        id=lancamento.id,
        data_hora=lancamento.data_hora,
        tipo=lancamento.tipo,
        valor=lancamento.valor,
        saldo_apos=lancamento.saldo_apos,
        motivo=lancamento.motivo,
        observacao=lancamento.observacao,
        usuario=usuario,
    )


def lancar_manual(
    s: Session,
    ator: Ator,
    cliente_id: int,
    tipo: TipoLancamento,
    valor: Decimal,
    motivo: MotivoLancamento,
    observacao: str = "",
) -> LinhaLancamento:
    exigir_modulo(ator, Modulo.CLIENTES)
    cliente = carregar_cliente(s, cliente_id)
    if not cliente.ativo:
        raise ValorInvalido("Cliente inativo: reative antes de lançar.")
    return _linha(lancar(s, ator, cliente, tipo, valor, motivo, observacao), ator.nome)


def extrato(s: Session, ator: Ator, cliente_id: int, inicio: date, fim: date) -> Extrato:
    """Lançamentos de `inicio` 00:00 até o fim do dia `fim`, com o saldo anterior ao período."""
    exigir_modulo(ator, Modulo.CLIENTES)
    if fim < inicio:
        raise ValorInvalido("A data final é anterior à inicial.")
    cliente = carregar_cliente(s, cliente_id)
    comeco = datetime.combine(inicio, time.min)
    depois_do_fim = datetime.combine(fim + timedelta(days=1), time.min)
    do_cliente = ClienteLancamento.cliente_id == cliente.id
    anterior = s.scalar(
        select(ClienteLancamento.saldo_apos)
        .where(do_cliente, ClienteLancamento.data_hora < comeco)
        .order_by(ClienteLancamento.data_hora.desc(), ClienteLancamento.id.desc())
        .limit(1)
    )
    saldo_anterior = anterior if anterior is not None else Decimal("0.00")
    consulta = (
        select(ClienteLancamento, Usuario.nome.label("usuario"))
        .join(Usuario, Usuario.id == ClienteLancamento.usuario_id)
        .where(
            do_cliente,
            ClienteLancamento.data_hora >= comeco,
            ClienteLancamento.data_hora < depois_do_fim,
        )
        .order_by(ClienteLancamento.data_hora, ClienteLancamento.id)
    )
    linhas = tuple(_linha(row.ClienteLancamento, row.usuario) for row in s.execute(consulta))
    return Extrato(
        cliente=resumo_cliente(cliente),
        inicio=inicio,
        fim=fim,
        saldo_anterior=saldo_anterior,
        linhas=linhas,
        saldo_final=linhas[-1].saldo_apos if linhas else saldo_anterior,
    )
```

```python
"""Extrato do cliente em HTML para impressão via QTextDocument (função pura)."""

from datetime import datetime
from html import escape

from cfauto.clientes.models import TITULOS_MOTIVO_LANCAMENTO, TipoLancamento
from cfauto.clientes.schemas import Extrato, LinhaLancamento
from cfauto.clientes.validacao import formatar_celular, formatar_cpf, formatar_saldo
from cfauto.core.money import formatar_brl


def descrever_lancamento(linha: LinhaLancamento) -> str:
    titulo = TITULOS_MOTIVO_LANCAMENTO[linha.motivo]
    return f"{titulo} — {linha.observacao}" if linha.observacao else titulo


def _linha_html(linha: LinhaLancamento) -> str:
    debito = formatar_brl(linha.valor) if linha.tipo == TipoLancamento.DEBITO else ""
    credito = formatar_brl(linha.valor) if linha.tipo == TipoLancamento.CREDITO else ""
    return (
        "<tr>"
        f"<td>{linha.data_hora:%d/%m/%Y %H:%M}</td>"
        f"<td>{escape(descrever_lancamento(linha))}</td>"
        f'<td align="right">{debito}</td>'
        f'<td align="right">{credito}</td>'
        f'<td align="right">{escape(formatar_saldo(linha.saldo_apos))}</td>'
        "</tr>"
    )


def gerar_html_extrato(extrato: Extrato, loja: str, emitido_em: datetime) -> str:
    cliente = extrato.cliente
    cpf = f" · CPF {formatar_cpf(cliente.cpf)}" if cliente.cpf else ""
    if extrato.linhas:
        corpo = "".join(_linha_html(linha) for linha in extrato.linhas)
    else:
        corpo = '<tr><td colspan="5">Nenhum lançamento no período.</td></tr>'
    return (
        "<html><body>"
        f"<h2>{escape(loja)}</h2>"
        "<h3>Extrato do cliente</h3>"
        f"<p>{escape(cliente.codigo)} — {escape(cliente.nome)}{cpf}"
        f" · Celular {formatar_celular(cliente.celular)}</p>"
        f"<p>Período: {extrato.inicio:%d/%m/%Y} a {extrato.fim:%d/%m/%Y}</p>"
        f"<p>Saldo anterior: {escape(formatar_saldo(extrato.saldo_anterior))}</p>"
        '<table border="1" cellspacing="0" cellpadding="4" width="100%">'
        "<tr><th>Data</th><th>Descrição</th><th>Débito</th><th>Crédito</th><th>Saldo</th></tr>"
        f"{corpo}</table>"
        f"<p><b>Saldo final: {escape(formatar_saldo(extrato.saldo_final))}</b></p>"
        f"<p>Emitido em {emitido_em:%d/%m/%Y %H:%M}</p>"
        "</body></html>"
    )
```

```python
            juros_dias=self.juros_dias.value(),

        )

        return resposta == QMessageBox.StandardButton.Yes

    def _salvar_pelo_botao(self) -> None:
        self.salvar()

    def salvar(self, confirmar: Callable[[str], bool] | None = None) -> None:
        """Valida, avisa de CPF/celular repetido (salvar mesmo assim?) e grava."""
        perguntar = confirmar or self._perguntar
        try:
            dados = self.dados()
            with transacao(self._fabrica) as s:
                duplicados = verificar_duplicados(s, self._ator, dados, self.cliente_id)
            if duplicados and not perguntar(_texto_duplicados(duplicados)):
                return
            with transacao(self._fabrica) as s:
                if self.cliente_id is None:
                    novo_id = criar_cliente(s, self._ator, dados)
                else:
                    editar_cliente(s, self._ator, self.cliente_id, dados)
                    novo_id = self.cliente_id
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.cliente_id = novo_id
        self.accept()
```

```python
        for botao in (novo, editar, self.botao_situacao, self.botao_excluir):
            self._botoes.addWidget(botao)
        self._botoes.addStretch()
        self._botoes.addWidget(self.mostrar_inativos)
        layout = QVBoxLayout(self)
        layout.addWidget(self.busca)
        layout.addLayout(self._botoes)
        layout.addWidget(self.tabela)

    def recarregar(self) -> None:
        with transacao(self._fabrica) as s:
            self._clientes = buscar_clientes(
                s,
                self._ator,
                self.busca.text(),
                incluir_inativos=self.mostrar_inativos.isChecked(),
            )
        preencher_tabela(
            self.tabela,
            [
                [
                    c.codigo,
                    c.nome,
                    formatar_celular(c.celular),
                    formatar_saldo(c.saldo_devedor),
                    "Ilimitado" if c.disponivel is None else formatar_brl(c.disponivel),
                    "Ativo" if c.ativo else "Inativo",
                ]
                for c in self._clientes

        for linha, cliente in enumerate(self._clientes):
            if cliente.saldo_devedor > 0:
                self.tabela.item(linha, _COLUNA_SALDO).setForeground(QColor(COR_DEVEDOR))

        self.tabela.setCurrentCell(linha, 0)

        linha = self.tabela.currentRow()
        return self._clientes[linha] if 0 <= linha < len(self._clientes) else None

        if DialogoCliente(self._fabrica, self._ator, parent=self).exec():

        cliente = self.selecionado()
        if cliente is None:
            return
        with transacao(self._fabrica) as s:
            ficha = obter_cliente(s, self._ator, cliente.id)
        if DialogoCliente(self._fabrica, self._ator, ficha, self).exec():
            self.recarregar()

    def _editar_pelo_duplo_clique(self, _indice: QModelIndex) -> None:
        self._editar()

    def alternar_situacao(self) -> None:
        cliente = self.selecionado()
        if cliente is None:
            return
        try:
            with transacao(self._fabrica) as s:
                if cliente.ativo:
                    inativar_cliente(s, self._ator, cliente.id)
                else:
                    reativar_cliente(s, self._ator, cliente.id)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.recarregar()

    def _perguntar(self, texto: str) -> bool:
        resposta = QMessageBox.question(self, "CFAUTO", texto)
        return resposta == QMessageBox.StandardButton.Yes

    def _excluir_pelo_botao(self) -> None:
        self.excluir_selecionado(self._perguntar)

    def excluir_selecionado(self, confirmar: Callable[[str], bool]) -> None:
        cliente = self.selecionado()
        if cliente is None:
            return
        texto = f"Excluir o cliente {cliente.codigo} — {cliente.nome}? Não dá para desfazer."
        if not confirmar(texto):
            return
        try:
            with transacao(self._fabrica) as s:
                excluir_cliente(s, self._ator, cliente.id)
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.recarregar()

        cliente = self.selecionado()
        inativo = cliente is not None and not cliente.ativo
        self.botao_situacao.setText("Reativar" if inativo else "Inativar")
```

```python
"""Lançamento manual no razão do cliente: aumentar ou abater a dívida."""

from html import escape

from PySide6.QtWidgets import (
    QComboBox,
    QDialog,
    QDialogButtonBox,
    QFormLayout,
    QHBoxLayout,
    QLabel,
    QLineEdit,
    QRadioButton,
    QVBoxLayout,
    QWidget,
)
from sqlalchemy.orm import Session, sessionmaker

from cfauto.clientes.models import TITULOS_MOTIVO_LANCAMENTO, MotivoLancamento, TipoLancamento
from cfauto.clientes.razao import TAMANHO_OBSERVACAO, lancar_manual, saldo_apos
from cfauto.clientes.schemas import ResumoCliente
from cfauto.clientes.validacao import formatar_saldo
from cfauto.core.db import transacao
from cfauto.core.errors import ErroDominio, ValorInvalido
from cfauto.core.money import arredondar
from cfauto.core.permissoes import Ator
from cfauto.ui.comum import ler_decimal, mostrar_erro


class DialogoLancamento(QDialog):
    def __init__(
        self,
        fabrica: sessionmaker[Session],
        ator: Ator,
        cliente: ResumoCliente,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._cliente = cliente
        self.setWindowTitle(f"Lançamento — {cliente.codigo}")
        cabecalho = QLabel(
            f"<b>{escape(cliente.codigo)} — {escape(cliente.nome)}</b><br>"
            f"Saldo atual: {escape(formatar_saldo(cliente.saldo_devedor))}"
        )
        self.aumentar = QRadioButton("Aumentar dívida")
        self.abater = QRadioButton("Abater dívida")
        self.aumentar.setChecked(True)
        self.valor = QLineEdit()
        self.motivo = QComboBox()
        for motivo in MotivoLancamento:
            self.motivo.addItem(TITULOS_MOTIVO_LANCAMENTO[motivo], motivo)
        self.observacao = QLineEdit()
        self.observacao.setMaxLength(TAMANHO_OBSERVACAO)
        self.previa = QLabel()
        self.aumentar.toggled.connect(self._atualizar_previa)
        self.valor.textChanged.connect(self._atualizar_previa)

        tipos = QHBoxLayout()
        tipos.addWidget(self.aumentar)
        tipos.addWidget(self.abater)
        form = QFormLayout()
        form.addRow("", tipos)
        form.addRow("Valor", self.valor)
        form.addRow("Motivo", self.motivo)
        form.addRow("Observação", self.observacao)
        botoes = QDialogButtonBox(
            QDialogButtonBox.StandardButton.Save | QDialogButtonBox.StandardButton.Cancel
        )
        botoes.accepted.connect(self.salvar)
        botoes.rejected.connect(self.reject)
        layout = QVBoxLayout(self)
        layout.addWidget(cabecalho)
        layout.addLayout(form)
        layout.addWidget(self.previa)
        layout.addWidget(botoes)
        self._atualizar_previa()

    def tipo(self) -> TipoLancamento:
        return TipoLancamento.DEBITO if self.aumentar.isChecked() else TipoLancamento.CREDITO

    def _atualizar_previa(self) -> None:
        try:
            valor = arredondar(ler_decimal(self.valor.text(), "Valor"))
        except ValorInvalido:
            # Valor incompleto enquanto o usuário digita: sem prévia; o erro vem ao salvar.
            self.previa.setText("Saldo após: —")
            return
        novo = saldo_apos(self._cliente.saldo_devedor, self.tipo(), valor)
        self.previa.setText(f"Saldo após: {formatar_saldo(novo)}")

    def salvar(self) -> None:
        try:
            valor = ler_decimal(self.valor.text(), "Valor")
            with transacao(self._fabrica) as s:
                lancar_manual(
                    s,
                    self._ator,
                    self._cliente.id,
                    self.tipo(),
                    valor,
                    MotivoLancamento(self.motivo.currentData()),
                    self.observacao.text(),
                )
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        self.accept()
```

```python
"""Extrato do cliente: razão por período, com impressão (PDF pela impressora do Windows)."""

from datetime import date, timedelta
from typing import cast

from PySide6.QtCore import QDate
from PySide6.QtGui import QTextDocument
from PySide6.QtPrintSupport import QPrintPreviewDialog
from PySide6.QtWidgets import (
    QDateEdit,
    QDialog,
    QFormLayout,
    QHBoxLayout,
    QLabel,
    QPushButton,
    QVBoxLayout,
    QWidget,
)
from sqlalchemy.orm import Session, sessionmaker

from cfauto.clientes.extrato_html import descrever_lancamento, gerar_html_extrato
from cfauto.clientes.models import TipoLancamento
from cfauto.clientes.razao import extrato
from cfauto.clientes.schemas import Extrato
from cfauto.clientes.validacao import formatar_saldo
from cfauto.core import clock, config
from cfauto.core.db import transacao
from cfauto.core.errors import ErroDominio
from cfauto.core.money import formatar_brl
from cfauto.core.permissoes import Ator
from cfauto.ui.comum import criar_tabela, mostrar_erro, preencher_tabela

DIAS_PADRAO = 90
_COLUNAS = ["Data/hora", "Descrição", "Débito", "Crédito", "Saldo", "Usuário"]


def _campo_data(valor: date) -> QDateEdit:
    campo = QDateEdit(QDate(valor.year, valor.month, valor.day))
    campo.setCalendarPopup(True)
    campo.setDisplayFormat("dd/MM/yyyy")
    return campo


class DialogoExtrato(QDialog):
    def __init__(
        self,
        fabrica: sessionmaker[Session],
        ator: Ator,
        cliente_id: int,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._fabrica = fabrica
        self._ator = ator
        self._cliente_id = cliente_id
        self._extrato: Extrato | None = None
        self._loja = ""
        self.setWindowTitle("Extrato")
        self.resize(760, 480)
        hoje = clock.hoje()
        self.inicio = _campo_data(hoje - timedelta(days=DIAS_PADRAO))
        self.fim = _campo_data(hoje)
        atualizar = QPushButton("Atualizar")
        atualizar.clicked.connect(self.recarregar)
        imprimir = QPushButton("Imprimir extrato")
        imprimir.clicked.connect(self.imprimir)
        self.tabela = criar_tabela(_COLUNAS)
        self.saldo_anterior = QLabel()
        self.saldo_final = QLabel()

        periodo = QFormLayout()
        periodo.addRow("De", self.inicio)
        periodo.addRow("Até", self.fim)
        topo = QHBoxLayout()
        topo.addLayout(periodo)
        topo.addWidget(atualizar)
        topo.addStretch()
        topo.addWidget(imprimir)
        layout = QVBoxLayout(self)
        layout.addLayout(topo)
        layout.addWidget(self.saldo_anterior)
        layout.addWidget(self.tabela)
        layout.addWidget(self.saldo_final)
        self.recarregar()

    def recarregar(self) -> None:
        inicio = cast(date, self.inicio.date().toPython())
        fim = cast(date, self.fim.date().toPython())
        try:
            with transacao(self._fabrica) as s:
                self._extrato = extrato(s, self._ator, self._cliente_id, inicio, fim)
                self._loja = str(config.obter(s, "empresa.nome", "") or "")
        except ErroDominio as erro:
            mostrar_erro(self, erro)
            return
        dados = self._extrato
        self.setWindowTitle(f"Extrato — {dados.cliente.codigo} {dados.cliente.nome}")
        preencher_tabela(
            self.tabela,
            [
                [
                    f"{linha.data_hora:%d/%m/%Y %H:%M}",
                    descrever_lancamento(linha),
                    formatar_brl(linha.valor) if linha.tipo == TipoLancamento.DEBITO else "",
                    formatar_brl(linha.valor) if linha.tipo == TipoLancamento.CREDITO else "",
                    formatar_saldo(linha.saldo_apos),
                    linha.usuario,
                ]
                for linha in dados.linhas
            ],
        )
        self.saldo_anterior.setText(f"Saldo anterior: {formatar_saldo(dados.saldo_anterior)}")
        self.saldo_final.setText(f"Saldo final: {formatar_saldo(dados.saldo_final)}")

    def html(self) -> str:
        if self._extrato is None:
            return ""
        return gerar_html_extrato(self._extrato, self._loja, clock.agora())

    def imprimir(self) -> None:
        if self._extrato is None:
            return
        documento = QTextDocument(self)
        documento.setHtml(self.html())
        self._abrir_previa(documento)

    def _abrir_previa(self, documento: QTextDocument) -> None:
        previa = QPrintPreviewDialog(self)
        previa.paintRequested.connect(documento.print_)
        previa.exec()
```

Nota: no PySide6 o método de impressão do `QTextDocument` chama-se `print_` (com sublinhado). Se o mypy ou a execução reclamar, confira com `uv run python -c "from PySide6.QtGui import QTextDocument; print(hasattr(QTextDocument, 'print_'))"`.
